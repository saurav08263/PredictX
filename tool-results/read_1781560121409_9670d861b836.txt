    66→
    67→type TradeStep = "trade" | "placing" | "live" | "result"
    68→
    69→export function PredictScreen({ onNavigate }: { onNavigate: (t: Tab) => void }) {
    70→  const {
    71→    country,
    72→    fmt,
    73→    mounted,
    74→    prices,
    75→    priceHistory,
    76→    currentRound,
    77→    wsConnected,
    78→    balance,
    79→    activeBet,
    80→    lastResult,
    81→    placeBet,
    82→    subscribeToRound,
    83→    fetchRoundInfo,
    84→    winEvent,
    85→    clearWinEvent,
    86→    clearLastResult,
    87→    tier,
    88→    dailyPredictions,
    89→    dailyLimit,
    90→    activatePremium,
    91→  } = useCrypto()
    92→
    93→  // --- Trading selections ---
    94→  const [selectedCoin, setSelectedCoin] = useState<Coin>("BTCUSDT")
    95→  const [selectedDuration, setSelectedDuration] = useState<Duration>(30)
    96→  const [selectedAmount, setSelectedAmount] = useState<number>(100)
    97→  const [customAmount, setCustomAmount] = useState("")
    98→  const [showCustomInput, setShowCustomInput] = useState(false)
    99→  const [betPlacing, setBetPlacing] = useState(false)
   100→  const [showPremiumModal, setShowPremiumModal] = useState(false)
   101→
   102→  // Derive effective step from store state
   103→  const effectiveStep: TradeStep = lastResult
   104→    ? "result"
   105→    : activeBet
   106→      ? "live"
   107→      : betPlacing
   108→        ? "placing"
   109→        : "trade"
   110→
   111→  // When there's an active bet, derive coin/duration from it
   112→  const displayCoin: Coin = activeBet?.round?.coin ? (activeBet.round.coin as Coin) : selectedCoin
   113→  const displayDuration: Duration = activeBet?.round?.duration ? (activeBet.round.duration as Duration) : selectedDuration
   114→
   115→  const coinData = COINS.find((c) => c.id === displayCoin)!
   116→  const currentPrice = prices[displayCoin]
   117→  const history = priceHistory[displayCoin] || []
   118→
   119→  // Subscribe to round when coin/duration changes
   120→  useEffect(() => {
   121→    if (!mounted) return
   122→    subscribeToRound(selectedCoin, selectedDuration)
   123→    fetchRoundInfo(selectedCoin, selectedDuration)
   124→  }, [mounted, selectedCoin, selectedDuration, subscribeToRound, fetchRoundInfo])
   125→
   126→  // Compute bet amount
   127→  const betAmount = customAmount ? Number(customAmount) : selectedAmount
   128→
   129→  // Potential payout
   130→  const potentialPayout = Math.round(betAmount * 1.8 * 100) / 100
   131→
   132→  // Price change for current coin
   133→  const basePrice = history.length > 5 ? history[0] : 0
   134→  const priceChange = basePrice > 0 && currentPrice > 0
   135→    ? ((currentPrice - basePrice) / basePrice * 100)
   136→    : 0
   137→  const isPriceUp = priceChange >= 0
   138→
   139→  // --- Place Bet instantly ---
   140→  const handlePlaceBet = useCallback(async (direction: "UP" | "DOWN") => {
   141→    if (betAmount <= 0 || betAmount > balance) return
   142→
   143→    if (tier === "FREE" && dailyLimit > 0 && dailyPredictions >= dailyLimit) {
   144→      setShowPremiumModal(true)
   145→      return
   146→    }
   147→
   148→    setBetPlacing(true)
   149→    try {
   150→      const result = await placeBet(direction, selectedCoin, selectedDuration, betAmount)
   151→      setBetPlacing(false)
   152→      if (!result.success && result.code === "DAILY_LIMIT_REACHED") {
   153→        setShowPremiumModal(true)
   154→      }
   155→    } catch {
   156→      setBetPlacing(false)
   157→    }
   158→  }, [betAmount, balance, tier, dailyLimit, dailyPredictions, placeBet, selectedCoin, selectedDuration])
   159→
   160→  const handlePlayAgain = () => {
   161→    clearWinEvent()
   162→    clearLastResult()
   163→    setCustomAmount("")
   164→  }
   165→
   166→  const getPriceChangeColor = (entry: number, exit: number) => {
   167→    if (exit > entry) return "text-bull"
   168→    if (exit < entry) return "text-bear"
   169→    return "text-muted-foreground"
   170→  }
   171→
   172→  // Coin tab color helper
   173→  const coinTabColor = (coin: Coin, isActive: boolean) => {
   174→    if (!isActive) return "border-transparent bg-muted/40 text-muted-foreground"
   175→    switch (coin) {
   176→      case "BTCUSDT": return "border-gold/50 bg-gold/10 text-gold shadow-[0_0_12px_rgba(245,197,24,0.12)]"
   177→      case "ETHUSDT": return "border-blue-500/50 bg-blue-500/10 text-blue-400 shadow-[0_0_12px_rgba(59,130,246,0.12)]"
   178→      case "SOLUSDT": return "border-purple-500/50 bg-purple-500/10 text-purple-400 shadow-[0_0_12px_rgba(168,85,247,0.12)]"
   179→    }
   180→  }
   181→
   182→  const coinIconBg = (coin: Coin) => {
   183→    switch (coin) {
   184→      case "BTCUSDT": return "bg-gold/20 text-gold"
   185→      case "ETHUSDT": return "bg-blue-500/20 text-blue-400"
   186→      case "SOLUSDT": return "bg-purple-500/20 text-purple-400"
   187→    }
   188→  }
   189→
   190→  const canBet = betAmount > 0 && betAmount <= balance && !betPlacing
   191→
   192→  // Format amount for chips
   193→  const fmtChip = (a: number) => a >= 1000 ? `${(a / 1000).toFixed(a % 1000 === 0 ? 0 : 1)}K` : String(a)
   194→
   195→  return (
   196→    <div className="flex flex-col gap-1.5 px-3 pb-3 pt-1.5">
   197→      {/* ── Daily limit / Premium badge ── */}
   198→      {tier === "FREE" && effectiveStep === "trade" && (
   199→        <div className="flex items-center justify-between rounded-lg border border-gold/30 bg-gold/5 px-3 py-1.5">
   200→          <div className="flex items-center gap-2">
   201→            <Zap className="size-3.5 text-gold" />
   202→            <span className="text-[10px] font-bold text-gold">
   203→              {dailyLimit > 0 ? `${Math.max(0, dailyLimit - dailyPredictions)}/${dailyLimit}` : "0/3"} FREE TODAY
   204→            </span>
   205→          </div>
   206→          <button
   207→            onClick={() => setShowPremiumModal(true)}
   208→            className="rounded-md bg-gold/20 px-2 py-0.5 text-[9px] font-bold text-gold hover:bg-gold/30 transition"
   209→          >
   210→            UNLIMITED →
   211→          </button>
   212→        </div>
   213→      )}
   214→      {tier === "PREMIUM" && effectiveStep === "trade" && (
   215→        <div className="flex items-center justify-center gap-1.5 rounded-lg border border-gold/40 bg-gold/10 px-3 py-1">
   216→          <Zap className="size-3 text-gold" />
   217→          <span className="text-[9px] font-bold text-gold">PREMIUM · UNLIMITED</span>
   218→        </div>
   219→      )}
   220→
   221→      <AnimatePresence mode="wait">
   222→        {/* ============================================================ */}
   223→        {/* ====== MAIN TRADING SCREEN ====== */}
   224→        {/* ============================================================ */}
   225→        {effectiveStep === "trade" && (
   226→          <motion.div
   227→            key="trade"
   228→            initial={{ opacity: 0 }}
   229→            animate={{ opacity: 1 }}
   230→            exit={{ opacity: 0 }}
   231→            className="flex flex-col gap-1.5"
   232→          >
   233→            {/* ── Coin Selector Pills ── */}
   234→            <div className="flex items-center gap-1.5">
   235→              {COINS.map((coin) => {
   236→                const isActive = selectedCoin === coin.id
   237→                const p = prices[coin.id]
   238→                const h = priceHistory[coin.id] || []
   239→                const bp = h.length > 5 ? h[0] : 0
   240→                const ch = bp > 0 && p > 0 ? ((p - bp) / bp * 100) : 0
   241→                const up = ch >= 0
   242→                return (
   243→                  <motion.button
   244→                    key={coin.id}
   245→                    onClick={() => setSelectedCoin(coin.id)}
   246→                    className={`flex flex-1 items-center gap-2 rounded-xl border px-2.5 py-2 transition-all ${coinTabColor(coin.id, isActive)}`}
   247→                    whileTap={{ scale: 0.96 }}
   248→                  >
   249→                    <span className={`flex size-7 items-center justify-center rounded-full text-xs font-bold ${isActive ? coinIconBg(coin.id) : "bg-muted text-muted-foreground"}`}>
   250→                      {coin.icon}
   251→                    </span>
   252→                    <div className="flex-1 text-left min-w-0">
   253→                      <p className="text-[11px] font-bold truncate">{coin.label}</p>
   254→                      <div className="flex items-center gap-1">
   255→                        <p className="text-[10px] font-extrabold tabular-nums truncate">
   256→                          {mounted && p > 0 ? `$${formatPrice(p, coin.id)}` : "--"}
   257→                        </p>
   258→                        {mounted && p > 0 && (
   259→                          <span className={`text-[8px] font-bold flex items-center ${up ? "text-bull" : "text-bear"}`}>
   260→                            {up ? <ChevronUp className="size-2" /> : <ChevronDown className="size-2" />}
   261→                            {Math.abs(ch).toFixed(1)}%
   262→                          </span>
   263→                        )}
   264→                      </div>
   265→                    </div>
   266→                  </motion.button>
   267→                )
   268→              })}
   269→            </div>
   270→
   271→            {/* ── Live Price + Chart ── */}
   272→            <div className="rounded-xl border border-border bg-card overflow-hidden">
   273→              {/* Price header */}
   274→              <div className="flex items-center justify-between px-3 pt-2.5 pb-0.5">
   275→                <div>
   276→                  <p className="text-[9px] font-semibold text-muted-foreground tracking-wider uppercase">Current Price</p>
   277→                  <div className="flex items-baseline gap-1.5">
   278→                    <p className="text-2xl font-extrabold tabular-nums leading-tight">
   279→                      {mounted && currentPrice > 0 ? `$${formatPrice(currentPrice, selectedCoin)}` : "--.--"}
   280→                    </p>
   281→                    <span className="text-[10px] font-semibold text-muted-foreground">USDT</span>
   282→                    {mounted && currentPrice > 0 && (
   283→                      <span className={`ml-1 flex items-center gap-0.5 rounded px-1.5 py-0.5 text-[9px] font-bold ${isPriceUp ? "bg-bull/15 text-bull" : "bg-bear/15 text-bear"}`}>
   284→                        {isPriceUp ? <ChevronUp className="size-2.5" /> : <ChevronDown className="size-2.5" />}
   285→                        {Math.abs(priceChange).toFixed(2)}%
   286→                      </span>
   287→                    )}
   288→                  </div>
   289→                </div>
   290→                <div className="flex items-center gap-1.5">
   291→                  {wsConnected && (
   292→                    <span className="flex items-center gap-1 rounded-md border border-bull/40 bg-bull/10 px-1.5 py-0.5 text-[8px] font-bold text-bull">
   293→                      <span className="size-1.5 rounded-full bg-bull animate-pulse" />LIVE
   294→                    </span>
   295→                  )}
   296→                  {currentRound && (
   297→                    <span className="flex items-center gap-1 rounded-md border border-border bg-muted px-1.5 py-0.5 text-[9px] font-bold tabular-nums">
   298→                      <Timer className="size-2.5 text-bull" />
   299→                      {formatClock(currentRound.timeLeft)}
   300→                    </span>
   301→                  )}
   302→                </div>
   303→              </div>
   304→
   305→              {/* Chart */}
   306→              {history.length > 2 && (
   307→                <div className="px-1 pb-1">
   308→                  <PriceChart
   309→                    points={history.slice(-40)}
   310→                    className="h-28 w-full"
   311→                    entryPrice={null}
   312→                    direction={null}
   313→                  />
   314→                </div>
   315→              )}
   316→            </div>
   317→
   318→            {/* ── Time Chips ── */}
   319→            <div>
   320→              <p className="text-[9px] font-bold tracking-[0.2em] text-muted-foreground mb-1 px-0.5 uppercase">Duration</p>
   321→              <div className="flex items-center gap-1 overflow-x-auto no-scrollbar">
   322→                {DURATIONS.map((d) => (
   323→                  <motion.button
   324→                    key={d.value}
   325→                    onClick={() => setSelectedDuration(d.value)}
   326→                    className={`shrink-0 rounded-lg border px-3.5 py-1.5 text-center transition-all ${
   327→                      selectedDuration === d.value
   328→                        ? "border-bull/50 bg-bull/15 text-bull shadow-[0_0_8px_rgba(47,224,122,0.15)]"
   329→                        : "border-border bg-card text-muted-foreground hover:border-bull/25"
   330→                    }`}
   331→                    whileTap={{ scale: 0.92 }}
   332→                  >
   333→                    <span className="text-xs font-extrabold">{d.label}</span>
   334→                  </motion.button>
   335→                ))}
   336→              </div>
   337→            </div>
   338→
   339→            {/* ── Amount Chips ── */}
   340→            <div>
   341→              <p className="text-[9px] font-bold tracking-[0.2em] text-muted-foreground mb-1 px-0.5 uppercase">Amount</p>
   342→              <div className="flex items-center gap-1 overflow-x-auto no-scrollbar">
   343→                {AMOUNT_OPTIONS.map((a) => (
   344→                  <motion.button
   345→                    key={a}
   346→                    onClick={() => { setSelectedAmount(a); setCustomAmount(""); setShowCustomInput(false) }}
   347→                    className={`shrink-0 rounded-lg border px-3.5 py-1.5 text-center transition-all ${
   348→                      selectedAmount === a && !customAmount && !showCustomInput
   349→                        ? "border-bull/50 bg-bull/15 text-bull shadow-[0_0_8px_rgba(47,224,122,0.15)]"
   350→                        : "border-border bg-card text-muted-foreground hover:border-bull/25"
   351→                    }`}
   352→                    whileTap={{ scale: 0.92 }}
   353→                  >
   354→                    <span className="text-xs font-bold">{country.symbol.trim()}{fmtChip(a)}</span>
   355→                  </motion.button>
   356→                ))}
   357→                <motion.button
   358→                  onClick={() => { setShowCustomInput(true); setCustomAmount("") }}
   359→                  className={`shrink-0 rounded-lg border px-3.5 py-1.5 text-center transition-all ${
   360→                    showCustomInput
   361→                      ? "border-bull/50 bg-bull/15 text-bull shadow-[0_0_8px_rgba(47,224,122,0.15)]"
   362→                      : "border-border bg-card text-muted-foreground hover:border-bull/25"
   363→                  }`}
   364→                  whileTap={{ scale: 0.92 }}
   365→                >
   366→                  <span className="text-xs font-bold">Custom</span>
   367→                </motion.button>
   368→              </div>
   369→
   370→              {/* Custom Amount Input */}
   371→              <AnimatePresence>
   372→                {showCustomInput && (
   373→                  <motion.div
   374→                    initial={{ opacity: 0, height: 0 }}
   375→                    animate={{ opacity: 1, height: "auto" }}
   376→                    exit={{ opacity: 0, height: 0 }}
   377→                    className="overflow-hidden mt-1.5"
   378→                  >
   379→                    <div className="flex items-center rounded-lg border border-border bg-card px-3 py-1.5 focus-within:border-bull/50 transition-colors">
   380→                      <span className="text-sm font-bold text-muted-foreground">{country.symbol.trim()}</span>
   381→                      <input
   382→                        autoFocus
   383→                        inputMode="numeric"
   384→                        value={customAmount}
   385→                        onChange={(e) => setCustomAmount(e.target.value.replace(/[^0-9]/g, ""))}
   386→                        placeholder="Enter amount"
   387→                        className="w-full bg-transparent px-2 text-base font-extrabold outline-none"
   388→                      />
   389→                      {customAmount && (
   390→                        <button onClick={() => setCustomAmount("")} className="text-muted-foreground hover:text-foreground">
   391→                          <X className="size-4" />
   392→                        </button>
   393→                      )}
   394→                    </div>
   395→                  </motion.div>
   396→                )}
   397→              </AnimatePresence>
   398→            </div>
   399→
   400→            {/* ── Payout Preview + Balance Bar ── */}
   401→            <div className="flex items-center gap-2">
   402→              <div className="flex-1 flex items-center justify-between rounded-xl bg-gradient-to-r from-bull/10 to-bull/5 border border-bull/20 px-3 py-2">
   403→                <div className="flex items-center gap-1.5">
   404→                  <TrendingUp className="size-3.5 text-bull" />
   405→                  <span className="text-[10px] text-muted-foreground font-semibold">Payout (1.8x)</span>
   406→                </div>
   407→                <span className="text-sm font-extrabold text-bull">{fmt(potentialPayout, 2)}</span>
   408→              </div>
   409→              <button
   410→                onClick={() => onNavigate("wallet")}
   411→                className="flex items-center gap-1.5 rounded-xl border border-border bg-card px-3 py-2 hover:border-bull/30 transition-colors"
   412→              >
   413→                <Wallet className="size-3.5 text-muted-foreground" />
   414→                <span className="text-sm font-bold">{fmt(balance, 2)}</span>
   415→                <Plus className="size-3 text-bull" />
   416→              </button>
   417→            </div>
   418→
   419→            {/* ── UP / DOWN — MASSIVE TRADING BUTTONS ── */}
   420→            <div className="grid grid-cols-2 gap-2.5 mt-0.5">
   421→              {/* UP Button */}
   422→              <motion.button
   423→                onClick={() => handlePlaceBet("UP")}
   424→                disabled={!canBet}
   425→                className="group relative flex flex-col items-center justify-center gap-0.5 rounded-2xl border-2 border-bull/50 bg-gradient-to-b from-bull/20 to-bull/5 py-5 transition-all hover:from-bull/30 hover:to-bull/10 hover:border-bull/70 disabled:opacity-30 disabled:hover:from-bull/20 disabled:hover:to-bull/5 disabled:hover:border-bull/50 active:scale-[0.97]"
   426→                whileTap={{ scale: canBet ? 0.95 : 1 }}
   427→              >
   428→                {/* Animated glow ring on hover */}
   429→                {canBet && (
   430→                  <span className="absolute inset-0 rounded-2xl opacity-0 group-hover:opacity-100 transition-opacity duration-300 shadow-[inset_0_0_30px_rgba(47,224,122,0.1),0_0_20px_rgba(47,224,122,0.15)]" />
   431→                )}
   432→                <ArrowUpRight className="size-10 text-bull relative z-10" />
   433→                <span className="text-xl font-black text-bull relative z-10 tracking-wide">UP</span>
   434→                <span className="text-[10px] font-semibold text-bull/60 relative z-10">Price will rise</span>
   435→                {canBet && betAmount > 0 && (
   436→                  <span className="text-[9px] font-bold text-bull/40 relative z-10 mt-0.5">
   437→                    Win {fmt(potentialPayout, 0)}
   438→                  </span>
   439→                )}
   440→              </motion.button>
   441→
   442→              {/* DOWN Button */}
   443→              <motion.button
   444→                onClick={() => handlePlaceBet("DOWN")}
   445→                disabled={!canBet}
   446→                className="group relative flex flex-col items-center justify-center gap-0.5 rounded-2xl border-2 border-bear/50 bg-gradient-to-b from-bear/20 to-bear/5 py-5 transition-all hover:from-bear/30 hover:to-bear/10 hover:border-bear/70 disabled:opacity-30 disabled:hover:from-bear/20 disabled:hover:to-bear/5 disabled:hover:border-bear/50 active:scale-[0.97]"
   447→                whileTap={{ scale: canBet ? 0.95 : 1 }}
   448→              >
   449→                {/* Animated glow ring on hover */}
   450→                {canBet && (
   451→                  <span className="absolute inset-0 rounded-2xl opacity-0 group-hover:opacity-100 transition-opacity duration-300 shadow-[inset_0_0_30px_rgba(242,63,75,0.1),0_0_20px_rgba(242,63,75,0.15)]" />
   452→                )}
   453→                <ArrowDownRight className="size-10 text-bear relative z-10" />
   454→                <span className="text-xl font-black text-bear relative z-10 tracking-wide">DOWN</span>
   455→                <span className="text-[10px] font-semibold text-bear/60 relative z-10">Price will fall</span>
   456→                {canBet && betAmount > 0 && (
   457→                  <span className="text-[9px] font-bold text-bear/40 relative z-10 mt-0.5">
   458→                    Win {fmt(potentialPayout, 0)}
   459→                  </span>
   460→                )}
   461→              </motion.button>
   462→            </div>
   463→
   464→            {/* Insufficient balance hint */}
   465→            {betAmount > balance && balance > 0 && (
   466→              <motion.p
   467→                initial={{ opacity: 0 }}
   468→                animate={{ opacity: 1 }}
   469→                className="text-center text-[10px] font-semibold text-bear"
   470→              >
   471→                Insufficient balance · <button onClick={() => onNavigate("wallet")} className="underline text-bull">Add funds</button>
   472→              </motion.p>
   473→            )}
   474→            {balance <= 0 && (
   475→              <motion.div

   834→function PremiumPlanButton({
   835→  label,
   836→  price,
   837→  plan,
   838→  badge,
   839→  onActivate,
   840→  onClose,
   841→}: {
   842→  label: string
   843→  price: string
   844→  plan: "monthly" | "yearly"
   845→  badge?: string
   846→  onActivate: (plan: "monthly" | "yearly") => Promise<{ success: boolean }>
   847→  onClose: () => void
   848→}) {
   849→  const [loading, setLoading] = useState(false)
   850→
   851→  const handleActivate = async () => {
   852→    setLoading(true)
   853→    const result = await onActivate(plan)
   854→    setLoading(false)
   855→    if (result.success) {
   856→      onClose()
   857→    }
   858→  }
   859→
   860→  return (
   861→    <motion.button
   862→      onClick={handleActivate}
   863→      disabled={loading}
   864→      className={`relative flex w-full items-center justify-between rounded-xl border px-4 py-3.5 text-left transition disabled:opacity-60 ${
   865→        badge ? "border-gold/50 bg-gold/10" : "border-border bg-card"
   866→      }`}
   867→      whileTap={{ scale: 0.97 }}
   868→    >
   869→      {badge && (
   870→        <span className="absolute -top-2 right-3 rounded-md bg-gold px-2 py-0.5 text-[9px] font-bold text-black">
   871→          {badge}
   872→        </span>
   873→      )}
   874→      <div>
   875→        <p className="font-bold">{label}</p>
   876→        <p className="text-xs text-muted-foreground">Cancel anytime</p>
   877→      </div>
   878→      <div className="flex items-center gap-2">
   879→        <span className="text-lg font-extrabold text-bull">{price}</span>
   880→        {loading && <span className="size-4 animate-spin rounded-full border-2 border-gold border-t-transparent" />}
   881→      </div>
   882→    </motion.button>
   883→  )
   884→}
