  1188→  1188→  const lo = min - pad

  1190→  1190→  const labels: string[] = []
  1191→  1191→  for (let i = 0; i < count; i++) {
  1192→  1192→    const v = hi - ((hi - lo) / (count - 1)) * i

  1194→  1194→  }
