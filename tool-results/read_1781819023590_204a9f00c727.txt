   137→type Store = {
   138→  // User
   139→  userId: string | null
   140→  userName: string
   141→  email: string | null
   142→  phone: string | null
   143→  country: Country
   144→  setCountry: (c: Country) => void
   145→  fmt: (n: number, decimals?: number) => string
   146→

   148→  authStatus: "loading" | "authenticated" | "unauthenticated"
   149→  authMethod: string | null
   150→  authLoading: boolean
   151→  authError: string | null
   152→  login: (email: string, password: string) => Promise<{ success: boolean; error?: string }>

   154→  loginWithGoogle: () => Promise<{ success: boolean; error?: string }>

   156→  verifyOtp: (phone: string, code: string) => Promise<{ success: boolean; error?: string }>
   157→  logout: () => Promise<void>
   158→
   159→  // Prices (from WebSocket or REST fallback)
   160→  prices: Record<Coin, number>
   161→  priceHistory: Record<Coin, number[]>
   162→  mounted: boolean
   163→  wsConnected: boolean
   164→
   165→  // Round info (from WebSocket or REST)
   166→  currentRound: RoundInfo | null
   167→
   168→  // User financials (from API)
   169→  balance: number
   170→  winnings: number
   171→  profit: number
   172→  bonus: number
   173→  wins: number
   174→  totalBets: number
   175→  winRate: number
   176→
   177→  // Active bet (from API)
   178→  activeBet: ActiveBet
   179→
   180→  // Last round result (from WebSocket round:end or REST polling)
   181→  lastResult: RoundResult
   182→
   183→  // Bet history (from API)
   184→  recentBets: BetHistory[]
   185→
   186→  // Transactions (from API)
   187→  transactions: Transaction[]
   188→
   189→  // Actions
   190→  placeBet: (direction: "UP" | "DOWN", coin: Coin, duration: Duration, amount: number) => Promise<{ success: boolean; error?: string; code?: string }>
   191→  deposit: (n: number, method?: string) => Promise<void>
   192→  withdraw: (n: number, method?: string) => Promise<void>
   193→  refreshUser: () => Promise<void>
   194→  refreshBets: () => Promise<void>
   195→  refreshTransactions: () => Promise<void>
   196→  fetchRoundInfo: (coin: Coin, duration: Duration) => Promise<RoundInfo | null>
   197→
   198→  // Win celebration
   199→  winEvent: WinEvent
   200→  clearWinEvent: () => void
   201→
   202→  // Clear last result
   203→  clearLastResult: () => void
   204→
   205→  // Toasts
   206→  toasts: Toast[]
   207→}

   212→function useHydrated(): boolean {

   222→export function CryptoProvider({ children }: { children: ReactNode }) {
   223→  const mounted = useHydrated()
   224→  const [country, setCountry] = useState<Country>(COUNTRIES[0])
   225→
   226→  // Auth state
   227→  const [authStatus, setAuthStatus] = useState<"loading" | "authenticated" | "unauthenticated">("loading")
   228→  const [authMethod, setAuthMethod] = useState<string | null>(null)
   229→  const [authLoading, setAuthLoading] = useState(false)
   230→  const [authError, setAuthError] = useState<string | null>(null)
   231→
   232→  // User
   233→  const [userId, setUserId] = useState<string | null>(null)
   234→  const [userName, setUserName] = useState("Trader")
   235→  const [email, setEmail] = useState<string | null>(null)
   236→  const [phone, setPhone] = useState<string | null>(null)
   237→  const [balance, setBalance] = useState(0)
   238→  const [winnings, setWinnings] = useState(0)
   239→  const [profit, setProfit] = useState(0)
   240→  const [bonus, setBonus] = useState(0)
   241→  const [wins, setWins] = useState(0)
   242→  const [totalBets, setTotalBets] = useState(0)
   243→  const [winRate, setWinRate] = useState(0)
   244→
   245→  // Prices from WebSocket or REST
   246→  const [prices, setPrices] = useState<Record<Coin, number>>({
   247→    BTCUSDT: 0,
   248→    ETHUSDT: 0,
   249→    SOLUSDT: 0,
   250→  })
   251→  const [priceHistory, setPriceHistory] = useState<Record<Coin, number[]>>({
   252→    BTCUSDT: [],
   253→    ETHUSDT: [],
   254→    SOLUSDT: [],
   255→  })
   256→  const [currentRound, setCurrentRound] = useState<RoundInfo | null>(null)
   257→  const [wsConnected, setWsConnected] = useState(false)
   258→
   259→  // Bet state
   260→  const [activeBet, setActiveBet] = useState<ActiveBet>(null)
   261→  const [recentBets, setRecentBets] = useState<BetHistory[]>([])
   262→  const [transactions, setTransactions] = useState<Transaction[]>([])
   263→
   264→  // Round result (entry vs exit price comparison)
   265→  const [lastResult, setLastResult] = useState<RoundResult>(null)
   266→  const clearLastResult = useCallback(() => setLastResult(null), [])
   267→
   268→  // Win event
   269→  const [winEvent, setWinEvent] = useState<WinEvent>(null)
   270→  const clearWinEvent = useCallback(() => setWinEvent(null), [])
   271→
   272→  // Toasts
   273→  const [toasts, setToasts] = useState<Toast[]>([])
   274→  const pushToast = useCallback((msg: string, tone: Toast["tone"]) => {
   275→    const id = Date.now() + Math.random()
   276→    setToasts((t) => [...t, { id, msg, tone }])
   277→    setTimeout(() => setToasts((t) => t.filter((x) => x.id !== id)), 3500)
   278→  }, [])
   279→
   280→  // Refs to avoid stale closures
   281→  const socketRef = useRef<Socket | null>(null)
   282→  const roundSubRef = useRef<string>("")
   283→  const activeBetRef = useRef<ActiveBet>(null)
   284→  const userIdRef = useRef<string | null>(null)
   285→  const countryRef = useRef<Country>(country)
   286→  const wsConnectedRef = useRef(false)
   287→  const currentRoundRef = useRef<RoundInfo | null>(null)
   288→
   289→  // Keep refs in sync
   290→  useEffect(() => { activeBetRef.current = activeBet }, [activeBet])
   291→  useEffect(() => { userIdRef.current = userId }, [userId])
   292→  useEffect(() => { countryRef.current = country }, [country])
   293→  useEffect(() => { wsConnectedRef.current = wsConnected }, [wsConnected])
   294→  useEffect(() => { currentRoundRef.current = currentRound }, [currentRound])
   295→
   296→  // --- Helper: Update price history from any source ---
   297→  const updatePriceHistory = useCallback((newPrices: Record<Coin, number>) => {
   298→    setPriceHistory((prev) => {
   299→      const updated = { ...prev }
   300→      for (const coin of ["BTCUSDT", "ETHUSDT", "SOLUSDT"] as Coin[]) {
   301→        if (newPrices[coin] > 0) {
   302→          const arr = [...(prev[coin] || []), newPrices[coin]]
   303→          if (arr.length > MAX_HISTORY) arr.shift()
   304→          updated[coin] = arr
   305→        }
   306→      }
   307→      return updated
   308→    })
   309→  }, [])
   310→
   311→  // --- Initialize User (also detects auth status) ---
   312→  useEffect(() => {
   313→    if (!mounted) return
   314→    fetch("/api/user")
   315→      .then(async (r) => {
   316→        if (r.status === 401) {
   317→          // Not authenticated — show login screen
   318→          setAuthStatus("unauthenticated")
   319→          return null
   320→        }
   321→        if (!r.ok) {
   322→          // Server error — fall back to login screen rather than staying stuck on loader
   323→          setAuthStatus("unauthenticated")
   324→          return null
   325→        }
   326→        return r.json()
   327→      })
   328→      .then((data) => {
   329→        if (!data) return
   330→        setUserId(data.id)
   331→        setUserName(data.name)
   332→        setEmail(data.email || null)
   333→        setPhone(data.phone || null)
   334→        setBalance(data.balance)
   335→        setWinnings(data.winnings)
   336→        setProfit(data.profit)
   337→        setBonus(data.bonus)
   338→        setWins(data.wins)
   339→        setTotalBets(data.totalBets)
   340→        setWinRate(data.winRate)
   341→        setAuthMethod(data.authMethod || null)
   342→        setAuthStatus("authenticated")
   343→      })
   344→      .catch((e) => {
   345→        console.error(e)
   346→        setAuthStatus("unauthenticated")
   347→      })
   348→  }, [mounted])
   349→
   350→  // --- Auth methods ---
   351→  // Internal: reload user data after auth change (must be defined BEFORE auth methods that use it)
   352→  const refreshUserInternal = useCallback(async () => {
   353→    try {
   354→      const res = await fetch("/api/user")
   355→      if (!res.ok) return
   356→      const data = await res.json()
   357→      setUserId(data.id)
   358→      setUserName(data.name)
   359→      setEmail(data.email || null)
   360→      setPhone(data.phone || null)
   361→      setBalance(data.balance)
   362→      setWinnings(data.winnings)
   363→      setProfit(data.profit)
   364→      setBonus(data.bonus)
   365→      setWins(data.wins)
   366→      setTotalBets(data.totalBets)
   367→      setWinRate(data.winRate)
   368→      setAuthMethod(data.authMethod || null)
   369→    } catch (e) {
   370→      console.error(e)
   371→    }
   372→  }, [])
   373→
   374→  const login = useCallback(async (email: string, password: string) => {
   375→    setAuthLoading(true)
   376→    setAuthError(null)
   377→    try {
   378→      const res = await fetch("/api/auth/login", {
   379→        method: "POST",
   380→        headers: { "Content-Type": "application/json" },
   381→        body: JSON.stringify({ email, password }),
   382→      })
   383→      const data = await res.json()
   384→      if (!res.ok) {
   385→        setAuthError(data.error || "Login failed")
   386→        return { success: false, error: data.error }
   387→      }
   388→      // Reload full user data
   389→      await refreshUserInternal()
   390→      setAuthMethod(data.authMethod || "email")
   391→      setAuthStatus("authenticated")
   392→      return { success: true }
   393→    } catch {
   394→      setAuthError("Network error — please try again")
   395→      return { success: false, error: "Network error" }
   396→    } finally {
   397→      setAuthLoading(false)
   398→    }
   399→  }, [refreshUserInternal])
   400→
   401→  const signup = useCallback(async (name: string, email: string, password: string) => {
   402→    setAuthLoading(true)
   403→    setAuthError(null)
   404→    try {
   405→      const res = await fetch("/api/auth/signup", {
   406→        method: "POST",
   407→        headers: { "Content-Type": "application/json" },
   408→        body: JSON.stringify({ name, email, password }),
   409→      })
   410→      const data = await res.json()
   411→      if (!res.ok) {
   412→        setAuthError(data.error || "Signup failed")
   413→        return { success: false, error: data.error }
   414→      }
   415→      await refreshUserInternal()
   416→      setAuthMethod(data.authMethod || "email")
   417→      setAuthStatus("authenticated")
   418→      return { success: true }
   419→    } catch {
   420→      setAuthError("Network error — please try again")
   421→      return { success: false, error: "Network error" }
   422→    } finally {
   423→      setAuthLoading(false)
   424→    }
   425→  }, [refreshUserInternal])
   426→
   427→  const loginWithGoogle = useCallback(async () => {
   428→    setAuthLoading(true)
   429→    setAuthError(null)
   430→    try {
   431→      const res = await fetch("/api/auth/google", {
   432→        method: "POST",
   433→        headers: { "Content-Type": "application/json" },
   434→        body: JSON.stringify({}),
   435→      })
   436→      const data = await res.json()
   437→      if (!res.ok) {
   438→        setAuthError(data.error || "Google sign-in failed")
   439→        return { success: false, error: data.error }
   440→      }
   441→      await refreshUserInternal()
   442→      setAuthMethod("google")
   443→      setAuthStatus("authenticated")
   444→      return { success: true }
   445→    } catch {
   446→      setAuthError("Network error — please try again")
   447→      return { success: false, error: "Network error" }
   448→    } finally {
   449→      setAuthLoading(false)
   450→    }
   451→  }, [refreshUserInternal])
   452→
   453→  const sendOtp = useCallback(async (phone: string) => {
   454→    setAuthLoading(true)
   455→    setAuthError(null)
   456→    try {
   457→      const res = await fetch("/api/auth/otp", {
   458→        method: "POST",
   459→        headers: { "Content-Type": "application/json" },
   460→        body: JSON.stringify({ phone }),
   461→      })
   462→      const data = await res.json()
   463→      if (!res.ok) {
   464→        setAuthError(data.error || "Failed to send OTP")
   465→        return { success: false, error: data.error }
   466→      }
   467→      return { success: true, otp: data.otp as string }
   468→    } catch {
   469→      setAuthError("Network error — please try again")
   470→      return { success: false, error: "Network error" }
   471→    } finally {
   472→      setAuthLoading(false)
   473→    }
   474→  }, [])
   475→
   476→  const verifyOtp = useCallback(async (phone: string, code: string) => {
   477→    setAuthLoading(true)
   478→    setAuthError(null)
   479→    try {
   480→      const res = await fetch("/api/auth/otp", {
   481→        method: "PUT",
   482→        headers: { "Content-Type": "application/json" },
   483→        body: JSON.stringify({ phone, code }),
   484→      })
   485→      const data = await res.json()
   486→      if (!res.ok) {
   487→        setAuthError(data.error || "OTP verification failed")
   488→        return { success: false, error: data.error }
   489→      }
   490→      await refreshUserInternal()
   491→      setAuthMethod("phone")
   492→      setAuthStatus("authenticated")
   493→      return { success: true }
   494→    } catch {
   495→      setAuthError("Network error — please try again")
   496→      return { success: false, error: "Network error" }
   497→    } finally {
   498→      setAuthLoading(false)
   499→    }
   500→  }, [refreshUserInternal])
   501→
   502→  const logout = useCallback(async () => {
   503→    try {
   504→      await fetch("/api/auth/logout", { method: "POST" })
   505→    } catch {

   507→    }
   508→    // Reset all state
   509→    setUserId(null)
   510→    setUserName("Trader")
   511→    setEmail(null)
   512→    setPhone(null)
   513→    setBalance(0)
   514→    setWinnings(0)
   515→    setProfit(0)
   516→    setBonus(0)
   517→    setWins(0)
   518→    setTotalBets(0)
   519→    setWinRate(0)
   520→    setActiveBet(null)
   521→    setRecentBets([])
   522→    setTransactions([])
   523→    setLastResult(null)
   524→    setAuthMethod(null)
   525→    setAuthStatus("unauthenticated")

   527→
   528→  // --- Subscribe to a specific coin+duration round ---
   529→  const subscribeToRound = useCallback((coin: Coin, duration: Duration) => {
   530→    const key = `${coin}-${duration}`
   531→    roundSubRef.current = key
   532→    socketRef.current?.emit("round:subscribe", { coin, duration })
   533→  }, [])
   534→
   535→  // --- Fetch round info via REST API (works without WebSocket) ---
   536→  const fetchRoundInfo = useCallback(async (coin: Coin, duration: Duration): Promise<RoundInfo | null> => {
   537→    try {
   538→      const res = await fetch(`/api/round/current?coin=${coin}&duration=${duration}`)
   539→      if (res.ok) {
   540→        const data = await res.json()
   541→        const info: RoundInfo = {
   542→          roundId: data.roundId,
   543→          coin: data.coin as Coin,
   544→          duration: data.duration,
   545→          timeLeft: data.timeLeft,
   546→          entryPrice: data.entryPrice,
   547→          currentPrice: data.currentPrice || data.entryPrice,
   548→        }
   549→        // Update current round state
   550→        setCurrentRound(info)
   551→        return info
   552→      }
   553→    } catch {
   554→      // Silently fail
   555→    }
   556→    return null
   557→  }, [])
   558→
   559→  // --- Fetch active bet ---
   560→  const refreshActiveBet = useCallback(async () => {
   561→    const uid = userIdRef.current
   562→    if (!uid) return
   563→    try {
   564→      const res = await fetch(`/api/bet/active?userId=${uid}`)
   565→      if (res.ok) {
   566→        const data = await res.json()
   567→        setActiveBet(data.activeBet)
   568→        if (data.activeBet) {
   569→          subscribeToRound(
   570→            data.activeBet.round.coin as Coin,
   571→            data.activeBet.round.duration as Duration,
   572→          )
   573→          // Also try to fetch round info via REST
   574→          fetchRoundInfo(
   575→            data.activeBet.round.coin as Coin,
   576→            data.activeBet.round.duration as Duration,
   577→          )
   578→        }
   579→      }
   580→    } catch (err) {
   581→      console.error("Failed to refresh active bet:", err)
   582→    }
   583→  }, [subscribeToRound, fetchRoundInfo])
   584→
   585→  // --- Refresh user data ---
   586→  const refreshUser = useCallback(async () => {
   587→    const uid = userIdRef.current
   588→    if (!uid) return
   589→    try {
   590→      const res = await fetch("/api/user")
   591→      if (res.ok) {
   592→        const data = await res.json()
   593→        setBalance(data.balance)
   594→        setWinnings(data.winnings)
   595→        setProfit(data.profit)
   596→        setBonus(data.bonus)
   597→        setWins(data.wins)
   598→        setTotalBets(data.totalBets)
   599→        setWinRate(data.winRate)
   600→      }
   601→    } catch (err) {
   602→      console.error("Failed to refresh user:", err)
   603→    }
   604→  }, [])
   605→
   606→  // --- Refresh bet history ---
   607→  const refreshBets = useCallback(async () => {
   608→    const uid = userIdRef.current

  1101→export function timeLabels(count = 6) {
  1102→  const now = new Date()
  1103→  const labels: string[] = []
  1104→  for (let i = count - 1; i >= 0; i--) {
  1105→    const d = new Date(now.getTime() - i * 60_000)
  1106→    labels.push(`${String(d.getHours()).padStart(2, "0")}:${String(d.getMinutes()).padStart(2, "0")}`)
  1107→  }
  1108→  return labels
  1109→}

  1111→export function priceLabels(history: number[], count = 7) {
  1112→  if (history.length === 0) return Array.from({ length: count }, () => "--.--")
  1113→  const min = Math.min(...history)
  1114→  const max = Math.max(...history)
  1115→  const pad = (max - min) * 0.1 || 100
  1116→  const lo = min - pad
  1117→  const hi = max + pad
  1118→  const labels: string[] = []
  1119→  for (let i = 0; i < count; i++) {
  1120→    const v = hi - ((hi - lo) / (count - 1)) * i
  1121→    labels.push(v.toLocaleString("en-US", { minimumFractionDigits: 2, maximumFractionDigits: 2 }))
  1122→  }
  1123→  return labels
  1124→}
