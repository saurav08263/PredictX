   137→type Store = {
   138→  // User
   139→  userId: string | null
   140→  userName: string

   169→  profit: number

   178→  dailyLimit: number // -1 means unlimited (premium)
   179→  premiumExpiresAt: string | null
   180→  activatePremium: (plan: "monthly" | "yearly") => Promise<{ success: boolean; error?: string }>
   181→  cancelPremium: () => Promise<void>
   182→
   183→  // Active bet (from API)
   184→  activeBet: ActiveBet
   185→
   186→  // Last round result (from WebSocket round:end or REST polling)
   187→  lastResult: RoundResult
   188→
   189→  // Bet history (from API)
   190→  recentBets: BetHistory[]
   191→
   192→  // Transactions (from API)
   193→  transactions: Transaction[]
   194→
   195→  // Actions
   196→  placeBet: (direction: "UP" | "DOWN", coin: Coin, duration: Duration, amount: number) => Promise<{ success: boolean; error?: string; code?: string }>
   197→  deposit: (n: number, method?: string) => Promise<void>
   198→  withdraw: (n: number, method?: string) => Promise<void>
   199→  refreshUser: () => Promise<void>
   200→  refreshBets: () => Promise<void>
   201→  refreshTransactions: () => Promise<void>
   202→  fetchRoundInfo: (coin: Coin, duration: Duration) => Promise<RoundInfo | null>
   203→
   204→  // Win celebration
   205→  winEvent: WinEvent
   206→  clearWinEvent: () => void
   207→
   208→  // Clear last result
   209→  clearLastResult: () => void
   210→
   211→  // Toasts
   212→  toasts: Toast[]
   213→}

   262→    BTCUSDT: [],

   265→  })
   266→  const [currentRound, setCurrentRound] = useState<RoundInfo | null>(null)

   268→
   269→  // Bet state
   270→  const [activeBet, setActiveBet] = useState<ActiveBet>(null)
   271→  const [recentBets, setRecentBets] = useState<BetHistory[]>([])
   272→  const [transactions, setTransactions] = useState<Transaction[]>([])
   273→
   274→  // Round result (entry vs exit price comparison)

   310→      for (const coin of ["BTCUSDT", "ETHUSDT", "SOLUSDT"] as Coin[]) {

   312→          const arr = [...(prev[coin] || []), newPrices[coin]]
   313→          if (arr.length > MAX_HISTORY) arr.shift()

   315→        }

   318→    })

   350→        setAuthStatus("authenticated")
   351→      })
   352→      .catch((e) => {

   354→        setAuthStatus("unauthenticated")

   366→      setUserName(data.name)
   367→      setBalance(data.balance)
   368→      setWinnings(data.winnings)
   369→      setProfit(data.profit)
   370→      setBonus(data.bonus)
   371→      setWins(data.wins)
   372→      setTotalBets(data.totalBets)
   373→      setWinRate(data.winRate)
   374→      setTier(data.tier || "FREE")
   375→      setDailyPredictions(data.dailyPredictions || 0)
   376→      setDailyLimit(data.dailyLimit ?? 3)
   377→      setPremiumExpiresAt(data.premiumExpiresAt || null)

   388→      const res = await fetch("/api/auth/login", {
   389→        method: "POST",
   390→        headers: { "Content-Type": "application/json" },
   391→        body: JSON.stringify({ email, password }),
   392→      })

   394→      if (!res.ok) {
   395→        setAuthError(data.error || "Login failed")

   403→    } catch {
   404→      setAuthError("Network error — please try again")
   405→      return { success: false, error: "Network error" }
   406→    } finally {
   407→      setAuthLoading(false)
   408→    }
   409→  }, [refreshUserInternal])

  1173→export function timeLabels(count = 6) {
  1174→  const now = new Date()
  1175→  const labels: string[] = []
  1176→  for (let i = count - 1; i >= 0; i--) {
  1177→    const d = new Date(now.getTime() - i * 60_000)
  1178→    labels.push(`${String(d.getHours()).padStart(2, "0")}:${String(d.getMinutes()).padStart(2, "0")}`)
  1179→  }
  1180→  return labels
  1181→}

  1183→export function priceLabels(history: number[], count = 7) {
  1184→  if (history.length === 0) return Array.from({ length: count }, () => "--.--")
  1185→  const min = Math.min(...history)
  1186→  const max = Math.max(...history)
  1187→  const pad = (max - min) * 0.1 || 100
  1188→  const lo = min - pad
  1189→  const hi = max + pad
  1190→  const labels: string[] = []
  1191→  for (let i = 0; i < count; i++) {
  1192→    const v = hi - ((hi - lo) / (count - 1)) * i
  1193→    labels.push(v.toLocaleString("en-US", { minimumFractionDigits: 2, maximumFractionDigits: 2 }))
  1194→  }
  1195→  return labels
  1196→}
