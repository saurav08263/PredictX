     1→"use client"
     2→
     3→import { useState, useEffect, useCallback } from "react"
     4→import { motion, AnimatePresence } from "framer-motion"
     5→import {
     6→  ChevronUp,
     7→  ChevronDown,
     8→  ArrowUpRight,
     9→  ArrowDownRight,
    10→  Timer,
    11→  Plus,
    12→  TrendingUp,
    13→  Loader2,
    14→  ShieldCheck,
    15→  Zap,
    16→  X,
    17→  Wallet,
    18→} from "lucide-react"
    19→import { PriceChart } from "../price-chart"
    20→import {
    21→  useCrypto,
    22→  formatClock,
    23→  formatPrice,
    24→  COINS,
    25→  DURATIONS,
    26→  AMOUNT_OPTIONS,
    27→  type Coin,
    28→  type Duration,
    29→} from "../store"
    30→import type { Tab } from "../app-shell"
    31→
    32→/**
    33→ * Single-Screen Trading Experience — Redesigned
    34→ *
    35→ * ONE screen shows everything:
    36→ *   - Coin tabs (BTC/ETH/SOL) — compact horizontal pills
    37→ *   - Live chart + price — prominent
    38→ *   - Time chips (5s, 10s, 20s, 30s, 40s, 50s, 1m) — scrollable row
    39→ *   - Amount chips (₹100, ₹500, ₹1000, ₹5000, Custom) — scrollable row
    40→ *   - Large UP / DOWN buttons — MASSIVE, full-width, with glow
    41→ *   - Potential payout preview — inline between chips and buttons
    42→ *
    43→ * Flow:
    44→ *   User opens screen → Selects time → Selects amount → Presses UP or DOWN → Bet placed instantly
    45→ *   Maximum 3 taps.
    46→ *
    47→ * States:
    48→ *   - "trade" — Main trading screen (coin, chart, chips, UP/DOWN)
    49→ *   - "placing" — Bet is being processed (loading spinner)
    50→ *   - "live" — Active bet tracking (chart with entry line, countdown)
    51→ *   - "result" — Round result (win/lose)
    52→ */
    53→
    54→type TradeStep = "trade" | "placing" | "live" | "result"
    55→
    56→export function PredictScreen({ onNavigate }: { onNavigate: (t: Tab) => void }) {
    57→  const {
    58→    country,
    59→    fmt,
    60→    mounted,
    61→    prices,
    62→    priceHistory,
    63→    currentRound,
    64→    wsConnected,
    65→    balance,
    66→    activeBet,
    67→    lastResult,
    68→    placeBet,
    69→    subscribeToRound,
    70→    fetchRoundInfo,
    71→    winEvent,
    72→    clearWinEvent,
    73→    clearLastResult,
    74→    tier,
    75→    dailyPredictions,
    76→    dailyLimit,
    77→    activatePremium,
    78→  } = useCrypto()
    79→
    80→  // --- Trading selections ---
    81→  const [selectedCoin, setSelectedCoin] = useState<Coin>("BTCUSDT")
    82→  const [selectedDuration, setSelectedDuration] = useState<Duration>(30)
    83→  const [selectedAmount, setSelectedAmount] = useState<number>(100)
    84→  const [customAmount, setCustomAmount] = useState("")
    85→  const [showCustomInput, setShowCustomInput] = useState(false)
    86→  const [betPlacing, setBetPlacing] = useState(false)
    87→  const [showPremiumModal, setShowPremiumModal] = useState(false)
    88→
    89→  // Derive effective step from store state
    90→  const effectiveStep: TradeStep = lastResult
    91→    ? "result"
    92→    : activeBet
    93→      ? "live"
    94→      : betPlacing
    95→        ? "placing"
    96→        : "trade"
    97→
    98→  // When there's an active bet, derive coin/duration from it
    99→  const displayCoin: Coin = activeBet?.round?.coin ? (activeBet.round.coin as Coin) : selectedCoin
   100→  const displayDuration: Duration = activeBet?.round?.duration ? (activeBet.round.duration as Duration) : selectedDuration
   101→
   102→  const coinData = COINS.find((c) => c.id === displayCoin)!
   103→  const currentPrice = prices[displayCoin]
   104→  const history = priceHistory[displayCoin] || []
   105→
   106→  // Subscribe to round when coin/duration changes
   107→  useEffect(() => {
   108→    if (!mounted) return
   109→    subscribeToRound(selectedCoin, selectedDuration)
   110→    fetchRoundInfo(selectedCoin, selectedDuration)
   111→  }, [mounted, selectedCoin, selectedDuration, subscribeToRound, fetchRoundInfo])
   112→
   113→  // Compute bet amount
   114→  const betAmount = customAmount ? Number(customAmount) : selectedAmount
   115→
   116→  // Potential payout
   117→  const potentialPayout = Math.round(betAmount * 1.8 * 100) / 100
   118→
   119→  // Price change for current coin
   120→  const basePrice = history.length > 5 ? history[0] : 0
   121→  const priceChange = basePrice > 0 && currentPrice > 0
   122→    ? ((currentPrice - basePrice) / basePrice * 100)
   123→    : 0
   124→  const isPriceUp = priceChange >= 0
   125→
   126→  // --- Place Bet instantly ---
   127→  const handlePlaceBet = useCallback(async (direction: "UP" | "DOWN") => {
   128→    if (betAmount <= 0 || betAmount > balance) return
   129→
   130→    if (tier === "FREE" && dailyLimit > 0 && dailyPredictions >= dailyLimit) {
   131→      setShowPremiumModal(true)
   132→      return
   133→    }
   134→
   135→    setBetPlacing(true)
   136→    try {
   137→      const result = await placeBet(direction, selectedCoin, selectedDuration, betAmount)
   138→      setBetPlacing(false)
   139→      if (!result.success && result.code === "DAILY_LIMIT_REACHED") {
   140→        setShowPremiumModal(true)
   141→      }
   142→    } catch {
   143→      setBetPlacing(false)
   144→    }
   145→  }, [betAmount, balance, tier, dailyLimit, dailyPredictions, placeBet, selectedCoin, selectedDuration])
   146→
   147→  const handlePlayAgain = () => {
   148→    clearWinEvent()
   149→    clearLastResult()
   150→    setCustomAmount("")
   151→  }
   152→
   153→  const getPriceChangeColor = (entry: number, exit: number) => {
   154→    if (exit > entry) return "text-bull"
   155→    if (exit < entry) return "text-bear"
   156→    return "text-muted-foreground"
   157→  }
   158→
   159→  // Coin tab color helper
   160→  const coinTabColor = (coin: Coin, isActive: boolean) => {
   161→    if (!isActive) return "border-transparent bg-muted/50 text-muted-foreground"
   162→    switch (coin) {
   163→      case "BTCUSDT": return "border-gold/40 bg-gold/10 text-gold"
   164→      case "ETHUSDT": return "border-blue-500/40 bg-blue-500/10 text-blue-400"
   165→      case "SOLUSDT": return "border-purple-500/40 bg-purple-500/10 text-purple-400"
   166→    }
   167→  }
   168→
   169→  const coinIconBg = (coin: Coin) => {
   170→    switch (coin) {
   171→      case "BTCUSDT": return "bg-gold/20 text-gold"
   172→      case "ETHUSDT": return "bg-blue-500/20 text-blue-400"
   173→      case "SOLUSDT": return "bg-purple-500/20 text-purple-400"
   174→    }
   175→  }
   176→
   177→  const canBet = betAmount > 0 && betAmount <= balance && !betPlacing
   178→
   179→  return (
   180→    <div className="flex flex-col gap-2 px-3 pb-3 pt-2">
   181→      {/* Daily limit indicator for free users */}
   182→      {tier === "FREE" && effectiveStep === "trade" && (
   183→        <div className="flex items-center justify-between rounded-lg border border-gold/30 bg-gold/5 px-3 py-1.5">
   184→          <div className="flex items-center gap-2">
   185→            <Zap className="size-3.5 text-gold" />
   186→            <span className="text-[10px] font-bold text-gold">
   187→              {dailyLimit > 0 ? `${Math.max(0, dailyLimit - dailyPredictions)}/${dailyLimit}` : "0/3"} FREE TODAY
   188→            </span>
   189→          </div>
   190→          <button
   191→            onClick={() => setShowPremiumModal(true)}
   192→            className="rounded-md bg-gold/20 px-2 py-0.5 text-[9px] font-bold text-gold hover:bg-gold/30 transition"
   193→          >
   194→            UNLIMITED →
   195→          </button>
   196→        </div>
   197→      )}
   198→
   199→      {/* Premium badge for premium users */}
   200→      {tier === "PREMIUM" && effectiveStep === "trade" && (
   201→        <div className="flex items-center justify-center gap-1.5 rounded-lg border border-gold/40 bg-gold/10 px-3 py-1.5">
   202→          <Zap className="size-3 text-gold" />
   203→          <span className="text-[10px] font-bold text-gold">PREMIUM · UNLIMITED PREDICTIONS</span>
   204→        </div>
   205→      )}
   206→
   207→      <AnimatePresence mode="wait">
   208→        {/* ====== MAIN TRADING SCREEN ====== */}
   209→        {effectiveStep === "trade" && (
   210→          <motion.div
   211→            key="trade"
   212→            initial={{ opacity: 0 }}
   213→            animate={{ opacity: 1 }}
   214→            exit={{ opacity: 0 }}
   215→            className="flex flex-col gap-2"
   216→          >
   217→            {/* ── Coin Selector Tabs ── */}
   218→            <div className="flex items-center gap-1.5">
   219→              {COINS.map((coin) => {
   220→                const isActive = selectedCoin === coin.id
   221→                const p = prices[coin.id]
   222→                const h = priceHistory[coin.id] || []
   223→                const bp = h.length > 5 ? h[0] : 0
   224→                const ch = bp > 0 && p > 0 ? ((p - bp) / bp * 100) : 0
   225→                const up = ch >= 0
   226→                return (
   227→                  <motion.button
   228→                    key={coin.id}
   229→                    onClick={() => setSelectedCoin(coin.id)}
   230→                    className={`flex flex-1 items-center gap-2 rounded-xl border px-2.5 py-2 transition-all ${coinTabColor(coin.id, isActive)}`}
   231→                    whileTap={{ scale: 0.96 }}
   232→                  >
   233→                    <span className={`flex size-7 items-center justify-center rounded-full text-xs font-bold ${isActive ? coinIconBg(coin.id) : "bg-muted text-muted-foreground"}`}>
   234→                      {coin.icon}
   235→                    </span>
   236→                    <div className="flex-1 text-left min-w-0">
   237→                      <p className="text-[11px] font-bold truncate">{coin.label}</p>
   238→                      <div className="flex items-center gap-1">
   239→                        <p className="text-[10px] font-extrabold tabular-nums truncate">
   240→                          {mounted && p > 0 ? `$${formatPrice(p, coin.id)}` : "--"}
   241→                        </p>
   242→                        {mounted && p > 0 && (
   243→                          <span className={`text-[8px] font-bold flex items-center ${up ? "text-bull" : "text-bear"}`}>
   244→                            {up ? <ChevronUp className="size-2" /> : <ChevronDown className="size-2" />}
   245→                            {Math.abs(ch).toFixed(1)}%
   246→                          </span>
   247→                        )}
   248→                      </div>
   249→                    </div>
   250→                  </motion.button>
   251→                )
   252→              })}
   253→            </div>
   254→
   255→            {/* ── Live Price + Chart ── */}
   256→            <div className="rounded-xl border border-border bg-card overflow-hidden">
   257→              {/* Price header */}
   258→              <div className="flex items-center justify-between px-3 pt-2.5 pb-0.5">
   259→                <div>
   260→                  <p className="text-[9px] font-semibold text-muted-foreground tracking-wider uppercase">Current Price</p>
   261→                  <div className="flex items-baseline gap-1.5">
   262→                    <p className="text-2xl font-extrabold tabular-nums leading-tight">
   263→                      {mounted && currentPrice > 0 ? `$${formatPrice(currentPrice, selectedCoin)}` : "--.--"}
   264→                    </p>
   265→                    <span className="text-[10px] font-semibold text-muted-foreground">USDT</span>
   266→                    {mounted && currentPrice > 0 && (
   267→                      <span className={`ml-1 flex items-center gap-0.5 rounded px-1.5 py-0.5 text-[9px] font-bold ${isPriceUp ? "bg-bull/15 text-bull" : "bg-bear/15 text-bear"}`}>
   268→                        {isPriceUp ? <ChevronUp className="size-2.5" /> : <ChevronDown className="size-2.5" />}
   269→                        {Math.abs(priceChange).toFixed(2)}%
   270→                      </span>
   271→                    )}
   272→                  </div>
   273→                </div>
   274→                <div className="flex items-center gap-1.5">
   275→                  {wsConnected && (
   276→                    <span className="flex items-center gap-1 rounded-md border border-bull/40 bg-bull/10 px-1.5 py-0.5 text-[8px] font-bold text-bull">
   277→                      <span className="size-1.5 rounded-full bg-bull animate-pulse" />LIVE
   278→                    </span>
   279→                  )}
   280→                  {currentRound && (
   281→                    <span className="flex items-center gap-1 rounded-md border border-border bg-muted px-1.5 py-0.5 text-[9px] font-bold tabular-nums">
   282→                      <Timer className="size-2.5 text-bull" />
   283→                      {formatClock(currentRound.timeLeft)}
   284→                    </span>
   285→                  )}
   286→                </div>
   287→              </div>
   288→
   289→              {/* Chart */}
   290→              {history.length > 2 && (
   291→                <div className="px-1 pb-1">
   292→                  <PriceChart
   293→                    points={history.slice(-40)}
   294→                    className="h-28 w-full"
   295→                    entryPrice={null}
   296→                    direction={null}
   297→                  />
   298→                </div>
   299→              )}
   300→            </div>
   301→
   302→            {/* ── Time Chips ── */}
   303→            <div>
   304→              <p className="text-[9px] font-bold tracking-[0.2em] text-muted-foreground mb-1 px-0.5 uppercase">Duration</p>
   305→              <div className="flex items-center gap-1 overflow-x-auto no-scrollbar">
   306→                {DURATIONS.map((d) => (
   307→                  <motion.button
   308→                    key={d.value}
   309→                    onClick={() => setSelectedDuration(d.value)}
   310→                    className={`shrink-0 rounded-lg border px-3.5 py-1.5 text-center transition-all ${
   311→                      selectedDuration === d.value
   312→                        ? "border-bull/50 bg-bull/15 text-bull shadow-[0_0_8px_rgba(47,224,122,0.15)]"
   313→                        : "border-border bg-card text-muted-foreground hover:border-bull/25"
   314→                    }`}
   315→                    whileTap={{ scale: 0.92 }}
   316→                  >
   317→                    <span className="text-xs font-extrabold">{d.label}</span>
   318→                  </motion.button>
   319→                ))}
   320→              </div>
   321→            </div>
   322→
   323→            {/* ── Amount Chips ── */}
   324→            <div>
   325→              <p className="text-[9px] font-bold tracking-[0.2em] text-muted-foreground mb-1 px-0.5 uppercase">Amount</p>
   326→              <div className="flex items-center gap-1 overflow-x-auto no-scrollbar">
   327→                {AMOUNT_OPTIONS.map((a) => (
   328→                  <motion.button
   329→                    key={a}
   330→                    onClick={() => { setSelectedAmount(a); setCustomAmount(""); setShowCustomInput(false) }}
   331→                    className={`shrink-0 rounded-lg border px-3.5 py-1.5 text-center transition-all ${
   332→                      selectedAmount === a && !customAmount && !showCustomInput
   333→                        ? "border-bull/50 bg-bull/15 text-bull shadow-[0_0_8px_rgba(47,224,122,0.15)]"
   334→                        : "border-border bg-card text-muted-foreground hover:border-bull/25"
   335→                    }`}
   336→                    whileTap={{ scale: 0.92 }}
   337→                  >
   338→                    <span className="text-xs font-bold">{country.symbol.trim()}{a >= 1000 ? `${(a / 1000).toFixed(a % 1000 === 0 ? 0 : 1)}K` : a}</span>
   339→                  </motion.button>
   340→                ))}
   341→                <motion.button
   342→                  onClick={() => { setShowCustomInput(true); setCustomAmount("") }}
   343→                  className={`shrink-0 rounded-lg border px-3.5 py-1.5 text-center transition-all ${
   344→                    showCustomInput
   345→                      ? "border-bull/50 bg-bull/15 text-bull shadow-[0_0_8px_rgba(47,224,122,0.15)]"
   346→                      : "border-border bg-card text-muted-foreground hover:border-bull/25"
   347→                  }`}
   348→                  whileTap={{ scale: 0.92 }}
   349→                >
   350→                  <span className="text-xs font-bold">Custom</span>
   351→                </motion.button>
   352→              </div>
   353→
   354→              {/* Custom Amount Input */}
   355→              <AnimatePresence>
   356→                {showCustomInput && (
   357→                  <motion.div
   358→                    initial={{ opacity: 0, height: 0 }}
   359→                    animate={{ opacity: 1, height: "auto" }}
   360→                    exit={{ opacity: 0, height: 0 }}
   361→                    className="overflow-hidden mt-1.5"
   362→                  >
   363→                    <div className="flex items-center rounded-lg border border-border bg-card px-3 py-1.5 focus-within:border-bull/50 transition-colors">
   364→                      <span className="text-sm font-bold text-muted-foreground">{country.symbol.trim()}</span>
   365→                      <input
   366→                        autoFocus
   367→                        inputMode="numeric"
   368→                        value={customAmount}
   369→                        onChange={(e) => setCustomAmount(e.target.value.replace(/[^0-9]/g, ""))}
   370→                        placeholder="Enter amount"
   371→                        className="w-full bg-transparent px-2 text-base font-extrabold outline-none"
   372→                      />
   373→                      {customAmount && (
   374→                        <button onClick={() => setCustomAmount("")} className="text-muted-foreground hover:text-foreground">
   375→                          <X className="size-4" />
   376→                        </button>
   377→                      )}
   378→                    </div>
   379→                  </motion.div>
   380→                )}
   381→              </AnimatePresence>
   382→            </div>
   383→
   384→            {/* ── Payout Preview + Balance Bar ── */}
   385→            <div className="flex items-center gap-2">
   386→              <div className="flex-1 flex items-center justify-between rounded-xl bg-gradient-to-r from-bull/10 to-bull/5 border border-bull/20 px-3 py-2">
   387→                <div className="flex items-center gap-1.5">
   388→                  <TrendingUp className="size-3.5 text-bull" />
   389→                  <span className="text-[10px] text-muted-foreground font-semibold">Payout (1.8x)</span>
   390→                </div>
   391→                <span className="text-sm font-extrabold text-bull">{fmt(potentialPayout, 2)}</span>
   392→              </div>
   393→              <button
   394→                onClick={() => onNavigate("wallet")}
   395→                className="flex items-center gap-1.5 rounded-xl border border-border bg-card px-3 py-2 hover:border-bull/30 transition-colors"
   396→              >
   397→                <Wallet className="size-3.5 text-muted-foreground" />
   398→                <span className="text-sm font-bold">{fmt(balance, 2)}</span>
   399→                <Plus className="size-3 text-bull" />
   400→              </button>
   401→            </div>
   402→
   403→            {/* ── UP / DOWN Buttons — MASSIVE ── */}
   404→            <div className="grid grid-cols-2 gap-2.5 mt-0.5">
   405→              <motion.button
   406→                onClick={() => handlePlaceBet("UP")}
   407→                disabled={!canBet}
   408→                className="group relative flex flex-col items-center justify-center gap-1 rounded-2xl border-2 border-bull/50 bg-gradient-to-b from-bull/20 to-bull/5 py-6 transition-all hover:from-bull/30 hover:to-bull/10 hover:border-bull/70 disabled:opacity-30 disabled:hover:from-bull/20 disabled:hover:to-bull/5 disabled:hover:border-bull/50 active:scale-[0.97]"
   409→                whileTap={{ scale: 0.95 }}
   410→              >
   411→                {/* Glow effect */}
   412→                {canBet && (
   413→                  <span className="absolute inset-0 rounded-2xl opacity-0 group-hover:opacity-100 transition-opacity duration-300 shadow-[inset_0_0_30px_rgba(47,224,122,0.1),0_0_20px_rgba(47,224,122,0.15)]" />
   414→                )}
   415→                <ArrowUpRight className="size-10 text-bull relative z-10" />
   416→                <span className="text-xl font-black text-bull relative z-10 tracking-wide">UP</span>
   417→                <span className="text-[10px] font-semibold text-bull/60 relative z-10">Price will rise</span>
   418→              </motion.button>
   419→              <motion.button
   420→                onClick={() => handlePlaceBet("DOWN")}
   421→                disabled={!canBet}
   422→                className="group relative flex flex-col items-center justify-center gap-1 rounded-2xl border-2 border-bear/50 bg-gradient-to-b from-bear/20 to-bear/5 py-6 transition-all hover:from-bear/30 hover:to-bear/10 hover:border-bear/70 disabled:opacity-30 disabled:hover:from-bear/20 disabled:hover:to-bear/5 disabled:hover:border-bear/50 active:scale-[0.97]"
   423→                whileTap={{ scale: 0.95 }}
   424→              >
   425→                {/* Glow effect */}
   426→                {canBet && (
   427→                  <span className="absolute inset-0 rounded-2xl opacity-0 group-hover:opacity-100 transition-opacity duration-300 shadow-[inset_0_0_30px_rgba(242,63,75,0.1),0_0_20px_rgba(242,63,75,0.15)]" />
   428→                )}
   429→                <ArrowDownRight className="size-10 text-bear relative z-10" />
   430→                <span className="text-xl font-black text-bear relative z-10 tracking-wide">DOWN</span>
   431→                <span className="text-[10px] font-semibold text-bear/60 relative z-10">Price will fall</span>
   432→              </motion.button>
   433→            </div>
   434→
   435→            {/* Insufficient balance hint */}
   436→            {betAmount > balance && balance > 0 && (
   437→              <motion.p
   438→                initial={{ opacity: 0 }}
   439→                animate={{ opacity: 1 }}
   440→                className="text-center text-[10px] font-semibold text-bear"
   441→              >
   442→                Insufficient balance · <button onClick={() => onNavigate("wallet")} className="underline text-bull">Add funds</button>
   443→              </motion.p>
   444→            )}
   445→            {balance <= 0 && (
   446→              <motion.div
   447→                initial={{ opacity: 0 }}
   448→                animate={{ opacity: 1 }}
   449→                className="text-center"
   450→              >
   451→                <button
   452→                  onClick={() => onNavigate("wallet")}
   453→                  className="inline-flex items-center gap-1.5 rounded-xl bg-bull/15 border border-bull/30 px-4 py-2 text-xs font-bold text-bull hover:bg-bull/25 transition"
   454→                >
   455→                  <Wallet className="size-3.5" /> Add funds to start trading
   456→                </button>
   457→              </motion.div>
   458→            )}
   459→
   460→            {/* Fair play notice */}
   461→            <div className="flex items-center justify-center gap-1.5 text-[9px] text-muted-foreground/60 pt-0.5">
   462→              <ShieldCheck className="size-2.5" />
   463→              Server-verified · Fair play · 1.8x return
   464→            </div>
   465→          </motion.div>
   466→        )}
   467→
   468→        {/* ====== PLACING BET ====== */}
   469→        {effectiveStep === "placing" && (
   470→          <motion.div
   471→            key="placing"
   472→            initial={{ opacity: 0 }}
   473→            animate={{ opacity: 1 }}
   474→            className="flex flex-col items-center justify-center gap-4 py-16"
   475→          >
   476→            <div className="relative">
   477→              <div className="size-20 rounded-full border-4 border-bull/20 flex items-center justify-center">
   478→                <Loader2 className="size-8 animate-spin text-bull" />
   479→              </div>
   480→            </div>
   481→            <div className="text-center">
   482→              <h2 className="text-xl font-extrabold mt-1">Placing Bet...</h2>
   483→              <p className="text-xs text-muted-foreground mt-1">Securely processing your prediction</p>

   485→            <div className="flex items-center gap-2 text-[10px] text-muted-foreground">

   799→function PremiumPlanButton({
   800→  label,
   801→  price,
   802→  plan,
   803→  badge,
   804→  onActivate,
   805→  onClose,
   806→}: {
   807→  label: string
   808→  price: string
   809→  plan: "monthly" | "yearly"
   810→  badge?: string
   811→  onActivate: (plan: "monthly" | "yearly") => Promise<{ success: boolean }>
   812→  onClose: () => void
   813→}) {
   814→  const [loading, setLoading] = useState(false)
   815→
   816→  const handleActivate = async () => {
   817→    setLoading(true)
   818→    const result = await onActivate(plan)
   819→    setLoading(false)
   820→    if (result.success) {
   821→      onClose()
   822→    }
   823→  }
   824→
   825→  return (
   826→    <motion.button
   827→      onClick={handleActivate}
   828→      disabled={loading}
   829→      className={`relative flex w-full items-center justify-between rounded-xl border px-4 py-3.5 text-left transition disabled:opacity-60 ${
   830→        badge ? "border-gold/50 bg-gold/10" : "border-border bg-card"
   831→      }`}
   832→      whileTap={{ scale: 0.97 }}
   833→    >
   834→      {badge && (
   835→        <span className="absolute -top-2 right-3 rounded-md bg-gold px-2 py-0.5 text-[9px] font-bold text-black">
   836→          {badge}
   837→        </span>
   838→      )}
   839→      <div>
   840→        <p className="font-bold">{label}</p>
   841→        <p className="text-xs text-muted-foreground">Cancel anytime</p>
   842→      </div>
   843→      <div className="flex items-center gap-2">
   844→        <span className="text-lg font-extrabold text-bull">{price}</span>
   845→        {loading && <span className="size-4 animate-spin rounded-full border-2 border-gold border-t-transparent" />}
   846→      </div>
   847→    </motion.button>
   848→  )
   849→}
