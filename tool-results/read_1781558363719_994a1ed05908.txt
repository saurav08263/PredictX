     1→"use client"
     2→
     3→import { useState, useEffect, useCallback } from "react"
     4→import { motion, AnimatePresence } from "framer-motion"
     5→import {
     6→  ChevronUp,
     7→  ChevronDown,
     8→  ArrowUpRight,
     9→  ArrowDownRight,
    10→  Timer,
    11→  Check,
    12→  Zap,
    13→  Plus,
    14→  TrendingUp,
    15→  Clock,
    16→  Loader2,
    17→  ShieldCheck,
    18→} from "lucide-react"
    19→import { PriceChart } from "../price-chart"
    20→import {
    21→  useCrypto,
    22→  formatClock,
    23→  formatPrice,
    24→  COINS,
    25→  DURATIONS,
    26→  AMOUNT_OPTIONS,
    27→  type Coin,
    28→  type Duration,
    29→} from "../store"
    30→import type { Tab } from "../app-shell"
    31→
    32→/**
    33→ * 10-Step Prediction Flow:
    34→ * 1. Select Coin (BTC/ETH/SOL)
    35→ * 2. Select Timeframe (5s–60s)
    36→ * 3. View Live Price + Chart
    37→ * 4. Select Direction (UP/DOWN)
    38→ * 5. Select Bet Amount
    39→ * 6. Review & Confirm
    40→ * 7. Placing Bet (processing)
    41→ * 8. Live Round Tracking (countdown)
    42→ * 9. Round Result (price comparison)
    43→ * 10. Wallet Update & Play Again
    44→ */
    45→type Step = "coin" | "time" | "price" | "direction" | "amount" | "confirm" | "placing" | "live" | "result"
    46→
    47→const STEP_LABELS: Record<Step, string> = {
    48→  coin: "Select Coin",
    49→  time: "Select Time",
    50→  price: "Live Price",
    51→  direction: "Predict",
    52→  amount: "Set Amount",
    53→  confirm: "Confirm",
    54→  placing: "Processing",
    55→  live: "Live Round",
    56→  result: "Result",
    57→}
    58→
    59→const STEP_ORDER: Step[] = ["coin", "time", "price", "direction", "amount", "confirm"]
    60→const STEP_NUMBERS: Record<Step, number> = {
    61→  coin: 1, time: 2, price: 3, direction: 4, amount: 5, confirm: 6,
    62→  placing: 7, live: 8, result: 9,
    63→}
    64→
    65→export function PredictScreen({ onNavigate }: { onNavigate: (t: Tab) => void }) {
    66→  const {
    67→    country,
    68→    fmt,
    69→    mounted,
    70→    prices,
    71→    priceHistory,
    72→    currentRound,
    73→    wsConnected,
    74→    balance,
    75→    activeBet,
    76→    lastResult,
    77→    placeBet,
    78→    subscribeToRound,
    79→    fetchRoundInfo,
    80→    winEvent,
    81→    clearWinEvent,
    82→    clearLastResult,
    83→    tier,
    84→    dailyPredictions,
    85→    dailyLimit,
    86→    activatePremium,
    87→  } = useCrypto()
    88→
    89→  const [step, setStep] = useState<Step>("coin")
    90→  const [selectedCoin, setSelectedCoin] = useState<Coin>("BTCUSDT")
    91→  const [selectedDuration, setSelectedDuration] = useState<Duration>(60)
    92→  const [direction, setDirection] = useState<"UP" | "DOWN" | null>(null)
    93→  const [selectedAmount, setSelectedAmount] = useState<number>(100)
    94→  const [customAmount, setCustomAmount] = useState("")
    95→  const [betPlacing, setBetPlacing] = useState(false)
    96→  const [showPremiumModal, setShowPremiumModal] = useState(false)
    97→
    98→  // Derive step from store state reactively
    99→  const effectiveStep: Step = lastResult
   100→    ? "result"
   101→    : (activeBet && step !== "result" ? "live" : step)
   102→
   103→  // When there's an active bet, derive coin/duration from it
   104→  const displayCoin: Coin = activeBet?.round?.coin ? (activeBet.round.coin as Coin) : selectedCoin
   105→  const displayDuration: Duration = activeBet?.round?.duration ? (activeBet.round.duration as Duration) : selectedDuration
   106→
   107→  const coinData = COINS.find((c) => c.id === displayCoin)!
   108→  const currentPrice = prices[displayCoin]
   109→  const history = priceHistory[displayCoin] || []
   110→
   111→  // Subscribe to round when coin/duration changes
   112→  useEffect(() => {
   113→    if (!mounted) return
   114→    subscribeToRound(selectedCoin, selectedDuration)
   115→    fetchRoundInfo(selectedCoin, selectedDuration)
   116→  }, [mounted, selectedCoin, selectedDuration, subscribeToRound, fetchRoundInfo])
   117→
   118→  const handleSelectCoin = (coin: Coin) => {
   119→    setSelectedCoin(coin)
   120→    setStep("time")
   121→  }
   122→
   123→  const handleSelectTime = (d: Duration) => {
   124→    setSelectedDuration(d)
   125→    setStep("price")
   126→  }
   127→
   128→  const handleSelectDirection = (dir: "UP" | "DOWN") => {
   129→    setDirection(dir)
   130→    setStep("amount")
   131→  }
   132→
   133→  const betAmount = customAmount ? Number(customAmount) : selectedAmount
   134→
   135→  const handleConfirmBet = async () => {
   136→    if (!direction || betAmount <= 0) return
   137→
   138→    if (tier === "FREE" && dailyLimit > 0 && dailyPredictions >= dailyLimit) {
   139→      setShowPremiumModal(true)
   140→      return
   141→    }
   142→
   143→    setStep("placing")
   144→    setBetPlacing(true)
   145→    try {
   146→      const result = await placeBet(direction, selectedCoin, selectedDuration, betAmount)
   147→      setBetPlacing(false)
   148→      if (result.success) {
   149→        setStep("live")
   150→      } else if (result.code === "DAILY_LIMIT_REACHED") {
   151→        setShowPremiumModal(true)
   152→        setStep("confirm")
   153→      } else {
   154→        setStep("confirm")
   155→      }
   156→    } catch {
   157→      setBetPlacing(false)
   158→      setStep("confirm")
   159→    }
   160→  }
   161→
   162→  const handlePlayAgain = () => {
   163→    setDirection(null)
   164→    setCustomAmount("")
   165→    setSelectedAmount(100)
   166→    clearWinEvent()
   167→    clearLastResult()
   168→    setStep("coin")
   169→  }
   170→
   171→  const goBack = () => {
   172→    switch (step) {
   173→      case "time": setStep("coin"); break
   174→      case "price": setStep("time"); break
   175→      case "direction": setStep("price"); break
   176→      case "amount": setStep("direction"); break
   177→      case "confirm": setStep("amount"); break
   178→      default: setStep("coin"); break
   179→    }
   180→  }
   181→
   182→  const getPriceChangeColor = (entry: number, exit: number) => {
   183→    if (exit > entry) return "text-bull"
   184→    if (exit < entry) return "text-bear"
   185→    return "text-muted-foreground"
   186→  }
   187→
   188→  const currentStepNum = STEP_NUMBERS[effectiveStep] || 1
   189→
   190→  return (
   191→    <div className="flex flex-col gap-3 px-4 pb-4 pt-3">
   192→      {/* Daily limit indicator for free users */}
   193→      {tier === "FREE" && effectiveStep !== "result" && (
   194→        <div className="flex items-center justify-between rounded-lg border border-gold/30 bg-gold/5 px-3 py-2">
   195→          <div className="flex items-center gap-2">
   196→            <Zap className="size-3.5 text-gold" />
   197→            <span className="text-[10px] font-bold text-gold">
   198→              {dailyLimit > 0 ? `${Math.max(0, dailyLimit - dailyPredictions)}/${dailyLimit}` : "0/3"} FREE TODAY
   199→            </span>
   200→          </div>
   201→          <button
   202→            onClick={() => setShowPremiumModal(true)}
   203→            className="rounded-md bg-gold/20 px-2 py-0.5 text-[9px] font-bold text-gold hover:bg-gold/30 transition"
   204→          >
   205→            UNLIMITED →
   206→          </button>
   207→        </div>
   208→      )}
   209→
   210→      {/* Premium badge for premium users */}
   211→      {tier === "PREMIUM" && effectiveStep !== "result" && (
   212→        <div className="flex items-center justify-center gap-1.5 rounded-lg border border-gold/40 bg-gold/10 px-3 py-1.5">
   213→          <Zap className="size-3 text-gold" />
   214→          <span className="text-[10px] font-bold text-gold">PREMIUM · UNLIMITED PREDICTIONS</span>
   215→        </div>
   216→      )}
   217→
   218→      {/* Step indicator (10 steps) */}
   219→      {effectiveStep !== "result" && (
   220→        <div className="space-y-1.5">
   221→          <div className="flex items-center gap-1.5">
   222→            {STEP_ORDER.map((s, i) => {
   223→              const stepIdx = STEP_ORDER.indexOf(effectiveStep as Step)
   224→              const isActive = effectiveStep === s || (effectiveStep === "placing" && s === "confirm") || (effectiveStep === "live" && i <= 5)
   225→              const isPast = stepIdx > i || effectiveStep === "live" || effectiveStep === "result"
   226→              return (
   227→                <div
   228→                  key={s}
   229→                  className={`h-1.5 flex-1 rounded-full transition-all duration-300 ${
   230→                    isActive ? "bg-bull" : isPast ? "bg-bull/40" : "bg-border"
   231→                  }`}
   232→                />
   233→              )
   234→            })}
   235→          </div>
   236→          <p className="text-center text-[10px] font-semibold text-muted-foreground">
   237→            STEP {currentStepNum} OF 10 — {STEP_LABELS[effectiveStep]}
   238→          </p>
   239→        </div>
   240→      )}
   241→
   242→      <AnimatePresence mode="wait">
   243→        {/* ====== STEP 1: SELECT COIN ====== */}
   244→        {effectiveStep === "coin" && (
   245→          <motion.div
   246→            key="coin"
   247→            initial={{ opacity: 0, x: 20 }}
   248→            animate={{ opacity: 1, x: 0 }}
   249→            exit={{ opacity: 0, x: -20 }}
   250→            className="flex flex-col gap-4"
   251→          >
   252→            <div className="text-center">
   253→              <p className="text-[10px] font-semibold tracking-widest text-muted-foreground">STEP 1 OF 10</p>
   254→              <h2 className="text-xl font-extrabold mt-1">Select Coin</h2>
   255→              <p className="text-xs text-muted-foreground mt-1">Choose your trading pair</p>
   256→            </div>
   257→
   258→            <div className="flex flex-col gap-3">
   259→              {COINS.map((coin) => {
   260→                const price = prices[coin.id]
   261→                const coinHistory = priceHistory[coin.id] || []
   262→                // Use the first history point as the base price for comparison
   263→                const basePrice = coinHistory.length > 5 ? coinHistory[0] : 0
   264→                const change = basePrice > 0 && price > 0
   265→                  ? ((price - basePrice) / basePrice * 100)
   266→                  : 0
   267→                const isUp = change >= 0
   268→
   269→                return (
   270→                  <motion.button
   271→                    key={coin.id}
   272→                    onClick={() => handleSelectCoin(coin.id)}
   273→                    className={`flex items-center gap-4 rounded-xl border p-4 text-left transition-all ${
   274→                      selectedCoin === coin.id
   275→                        ? "border-bull/50 bg-bull/10"
   276→                        : "border-border bg-card hover:border-bull/30"
   277→                    }`}
   278→                    whileTap={{ scale: 0.97 }}
   279→                  >
   280→                    <span className={`flex size-12 items-center justify-center rounded-full text-xl font-bold ${
   281→                      coin.id === "BTCUSDT" ? "bg-gold/20 text-gold" :
   282→                      coin.id === "ETHUSDT" ? "bg-blue-500/20 text-blue-400" :
   283→                      "bg-purple-500/20 text-purple-400"
   284→                    }`}>
   285→                      {coin.icon}
   286→                    </span>
   287→                    <div className="flex-1">
   288→                      <p className="font-bold">{coin.label}</p>
   289→                      <p className="text-xs text-muted-foreground">USDT Pair</p>
   290→                    </div>
   291→                    <div className="text-right">
   292→                      <p className="font-extrabold tabular-nums">
   293→                        {mounted && price > 0 ? `$${formatPrice(price, coin.id)}` : "--.--"}
   294→                      </p>
   295→                      <p className={`text-xs font-semibold flex items-center justify-end gap-0.5 ${isUp ? "text-bull" : "text-bear"}`}>
   296→                        {mounted && price > 0 ? (
   297→                          <>
   298→                            {isUp ? <ChevronUp className="size-3" /> : <ChevronDown className="size-3" />}
   299→                            {Math.abs(change).toFixed(2)}%
   300→                          </>
   301→                        ) : "--.--%"}
   302→                      </p>
   303→                    </div>
   304→                  </motion.button>
   305→                )
   306→              })}
   307→            </div>
   308→
   309→            {wsConnected && (
   310→              <p className="flex items-center justify-center gap-1.5 text-[10px] text-bull">
   311→                <span className="size-1.5 rounded-full bg-bull animate-pulse" /> LIVE PRICES FROM BINANCE
   312→              </p>
   313→            )}
   314→          </motion.div>
   315→        )}
   316→
   317→        {/* ====== STEP 2: SELECT TIME ====== */}
   318→        {effectiveStep === "time" && (
   319→          <motion.div
   320→            key="time"
   321→            initial={{ opacity: 0, x: 20 }}
   322→            animate={{ opacity: 1, x: 0 }}
   323→            exit={{ opacity: 0, x: -20 }}
   324→            className="flex flex-col gap-4"
   325→          >
   326→            <button onClick={goBack} className="text-xs text-muted-foreground hover:text-foreground transition flex items-center gap-1">← Back</button>
   327→            <div className="text-center">
   328→              <p className="text-[10px] font-semibold tracking-widest text-muted-foreground">STEP 2 OF 10</p>
   329→              <h2 className="text-xl font-extrabold mt-1">Select Timeframe</h2>
   330→              <p className="text-xs text-muted-foreground mt-1">How long until the round ends?</p>
   331→            </div>
   332→
   333→            <div className="grid grid-cols-4 gap-2.5">
   334→              {DURATIONS.map((d) => (
   335→                <motion.button
   336→                  key={d.value}
   337→                  onClick={() => handleSelectTime(d.value)}
   338→                  className={`flex flex-col items-center gap-1.5 rounded-xl border py-4 transition-all ${
   339→                    selectedDuration === d.value
   340→                      ? "border-bull/50 bg-bull/10 text-bull"
   341→                      : "border-border bg-card hover:border-bull/30"
   342→                  }`}
   343→                  whileTap={{ scale: 0.95 }}
   344→                >
   345→                  <Clock className="size-5" />
   346→                  <span className="text-lg font-extrabold">{d.label}</span>
   347→                  <span className="text-[9px] font-semibold text-muted-foreground">{d.value}s ROUND</span>
   348→                </motion.button>
   349→              ))}
   350→            </div>
   351→
   352→            {/* Current price preview */}
   353→            <div className="rounded-xl border border-border bg-card p-3 text-center">
   354→              <p className="text-[10px] font-semibold text-muted-foreground">{coinData.label} CURRENT PRICE</p>
   355→              <p className="text-2xl font-extrabold tabular-nums mt-1">
   356→                {mounted && currentPrice > 0 ? `$${formatPrice(currentPrice, selectedCoin)}` : "--.--"}
   357→              </p>
   358→            </div>
   359→          </motion.div>
   360→        )}
   361→
   362→        {/* ====== STEP 3: VIEW LIVE PRICE ====== */}
   363→        {effectiveStep === "price" && (
   364→          <motion.div
   365→            key="price"
   366→            initial={{ opacity: 0, x: 20 }}
   367→            animate={{ opacity: 1, x: 0 }}
   368→            exit={{ opacity: 0, x: -20 }}
   369→            className="flex flex-col gap-3"
   370→          >
   371→            <button onClick={goBack} className="text-xs text-muted-foreground hover:text-foreground transition flex items-center gap-1">← Back</button>
   372→
   373→            <div className="text-center">
   374→              <p className="text-[10px] font-semibold tracking-widest text-muted-foreground">STEP 3 OF 10</p>
   375→              <h2 className="text-xl font-extrabold mt-1">Live Price</h2>
   376→              <p className="text-xs text-muted-foreground mt-1">Analyze the current market before predicting</p>
   377→            </div>
   378→
   379→            {/* Coin + Duration header */}
   380→            <div className="flex items-center justify-between">
   381→              <div className="flex items-center gap-2">
   382→                <span className={`flex size-8 items-center justify-center rounded-full text-sm font-bold ${
   383→                  selectedCoin === "BTCUSDT" ? "bg-gold/20 text-gold" :
   384→                  selectedCoin === "ETHUSDT" ? "bg-blue-500/20 text-blue-400" :
   385→                  "bg-purple-500/20 text-purple-400"
   386→                }`}>
   387→                  {coinData.icon}
   388→                </span>
   389→                <div>
   390→                  <p className="font-bold">{coinData.label}</p>
   391→                  <p className="text-[10px] text-muted-foreground">{selectedDuration}s round</p>
   392→                </div>
   393→              </div>
   394→              <span className="flex items-center gap-1 rounded-md border border-bull/40 bg-bull/10 px-2 py-1 text-xs font-bold text-bull">
   395→                <span className="size-1.5 rounded-full bg-bull animate-pulse" /> LIVE
   396→              </span>
   397→            </div>
   398→
   399→            {/* Big price display */}
   400→            <div className="rounded-2xl border border-border bg-card p-5 text-center">
   401→              <p className="text-[10px] font-semibold tracking-wide text-muted-foreground">CURRENT PRICE</p>
   402→              <p className="text-4xl font-extrabold tabular-nums mt-2">
   403→                {mounted && currentPrice > 0 ? `$${formatPrice(currentPrice, selectedCoin)}` : "--.--"}
   404→              </p>
   405→              <p className="text-xs text-muted-foreground mt-1">USDT</p>
   406→            </div>
   407→
   408→            {/* Chart */}
   409→            {history.length > 2 && (
   410→              <div className="rounded-xl border border-border bg-panel p-2">
   411→                <PriceChart
   412→                  points={history.slice(-40)}
   413→                  className="h-32 w-full"
   414→                  entryPrice={null}
   415→                  direction={null}
   416→                />
   417→              </div>
   418→            )}
   419→
   420→            {/* Round timer */}
   421→            {currentRound && (
   422→              <div className="flex items-center justify-between rounded-xl border border-bull/40 bg-bull/5 px-3 py-2">
   423→                <div className="flex items-center gap-2">
   424→                  <Timer className="size-4 text-bull" />
   425→                  <span className="text-xs font-bold">ROUND ENDS IN</span>
   426→                </div>
   427→                <span className="text-lg font-extrabold tabular-nums text-bull">
   428→                  {formatClock(currentRound.timeLeft)}
   429→                </span>
   430→              </div>
   431→            )}
   432→
   433→            {/* Continue to direction selection */}
   434→            <motion.button
   435→              onClick={() => setStep("direction")}
   436→              className="flex items-center justify-center gap-2 rounded-xl bg-bull py-3.5 text-sm font-bold text-black glow-green"
   437→              whileTap={{ scale: 0.97 }}
   438→            >
   439→              <TrendingUp className="size-4" /> CHOOSE DIRECTION →
   440→            </motion.button>
   441→          </motion.div>
   442→        )}
   443→
   444→        {/* ====== STEP 4: SELECT DIRECTION (UP/DOWN) ====== */}
   445→        {effectiveStep === "direction" && (
   446→          <motion.div
   447→            key="direction"
   448→            initial={{ opacity: 0, x: 20 }}
   449→            animate={{ opacity: 1, x: 0 }}
   450→            exit={{ opacity: 0, x: -20 }}
   451→            className="flex flex-col gap-3"
   452→          >
   453→            <button onClick={goBack} className="text-xs text-muted-foreground hover:text-foreground transition flex items-center gap-1">← Back</button>
   454→
   455→            <div className="text-center">
   456→              <p className="text-[10px] font-semibold tracking-widest text-muted-foreground">STEP 4 OF 10</p>
   457→              <h2 className="text-xl font-extrabold mt-1">Predict the Price</h2>
   458→              <p className="text-xs text-muted-foreground">Will it go UP or DOWN?</p>
   459→            </div>
   460→
   461→            {/* Quick price display */}
   462→            <div className="rounded-xl border border-border bg-card p-3 text-center">
   463→              <p className="text-[10px] font-semibold text-muted-foreground">{coinData.label}</p>
   464→              <p className="text-2xl font-extrabold tabular-nums">
   465→                {mounted && currentPrice > 0 ? `$${formatPrice(currentPrice, selectedCoin)}` : "--.--"}
   466→              </p>
   467→            </div>
   468→
   469→            {/* UP/DOWN buttons */}
   470→            <div className="grid grid-cols-2 gap-3">
   471→              <motion.button
   472→                onClick={() => handleSelectDirection("UP")}
   473→                className="flex flex-col items-center gap-2 rounded-2xl border-2 border-bull/30 bg-bull/10 py-8 transition-all hover:border-bull/60 hover:bg-bull/20"
   474→                whileTap={{ scale: 0.95 }}
   475→              >
   476→                <ArrowUpRight className="size-12 text-bull" />
   477→                <span className="text-2xl font-extrabold text-bull">UP</span>
   478→                <span className="text-[10px] font-semibold text-bull/70">Price will rise</span>
   479→              </motion.button>
   480→              <motion.button
   481→                onClick={() => handleSelectDirection("DOWN")}
   482→                className="flex flex-col items-center gap-2 rounded-2xl border-2 border-bear/30 bg-bear/10 py-8 transition-all hover:border-bear/60 hover:bg-bear/20"
   483→                whileTap={{ scale: 0.95 }}
   484→              >
   485→                <ArrowDownRight className="size-12 text-bear" />
   486→                <span className="text-2xl font-extrabold text-bear">DOWN</span>
   487→                <span className="text-[10px] font-semibold text-bear/70">Price will fall</span>
   488→              </motion.button>
   489→            </div>
   490→
   491→            {/* Fair play notice */}
   492→            <div className="flex items-center justify-center gap-1.5 text-[10px] text-muted-foreground">
   493→              <ShieldCheck className="size-3.5" />
   494→              Server-verified · Fair play guaranteed
   495→            </div>
   496→          </motion.div>
   497→        )}
   498→
   499→        {/* ====== STEP 5: SELECT AMOUNT ====== */}
   500→        {effectiveStep === "amount" && (
   501→          <motion.div
   502→            key="amount"
   503→            initial={{ opacity: 0, x: 20 }}
   504→            animate={{ opacity: 1, x: 0 }}
   505→            exit={{ opacity: 0, x: -20 }}
   506→            className="flex flex-col gap-4"
   507→          >
   508→            <button onClick={goBack} className="text-xs text-muted-foreground hover:text-foreground transition flex items-center gap-1">← Back</button>
   509→
   510→            <div className="text-center">
   511→              <p className="text-[10px] font-semibold tracking-widest text-muted-foreground">STEP 5 OF 10</p>
   512→              <h2 className="text-xl font-extrabold mt-1">Set Bet Amount</h2>
   513→            </div>
   514→
   515→            {/* Direction summary */}
   516→            <div className={`flex items-center justify-center gap-2 rounded-xl border py-3 ${
   517→              direction === "UP" ? "border-bull/40 bg-bull/10 text-bull" : "border-bear/40 bg-bear/10 text-bear"
