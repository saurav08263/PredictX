   355→function ReferContent({ userName, userId }: { userName: string; userId: string | null }) {
   356→  const referralCode = userId ? `CP${userId.slice(0, 6).toUpperCase()}` : "CPDEMO01"
   357→  const referralLink = `https://cryptopredictor.app/r/${referralCode}`
   358→  const [copied, setCopied] = useState(false)
   359→  const [shareStatus, setShareStatus] = useState<string | null>(null)
   360→
   361→  const handleCopy = () => {
   362→    navigator.clipboard?.writeText(referralLink).catch(() => {})
   363→    setCopied(true)
   364→    setTimeout(() => setCopied(false), 2000)
   365→  }
   366→
   367→  const handleShare = (platform: "whatsapp" | "telegram" | "more") => {
   368→    const text = `🚀 Join me on PredicTX and get ₹10,000 welcome bonus! Use my referral code ${referralCode}. ${referralLink}`
   369→    try {
   370→      if (platform === "whatsapp") {
   371→        window.open(`https://wa.me/?text=${encodeURIComponent(text)}`, "_blank")
   372→      } else if (platform === "telegram") {
   373→        window.open(`https://t.me/share/url?url=${encodeURIComponent(referralLink)}&text=${encodeURIComponent(text)}`, "_blank")
   374→      } else {
   375→        // Use native share sheet if available, otherwise copy
   376→        if (navigator.share) {
   377→          navigator.share({ title: "PredicTX", text, url: referralLink }).catch(() => {})
   378→        } else {
   379→          navigator.clipboard?.writeText(text).catch(() => {})
   380→          setShareStatus("Link copied to clipboard")
   381→          setTimeout(() => setShareStatus(null), 2000)
   382→        }
   383→      }
   384→    } catch {
   385→      setShareStatus("Unable to share — link copied instead")
   386→      navigator.clipboard?.writeText(referralLink).catch(() => {})
   387→      setTimeout(() => setShareStatus(null), 2000)
   388→    }
   389→  }
   390→
   391→  return (
   392→    <div className="space-y-4">
   393→      <div className="rounded-xl border border-gold/30 bg-gradient-to-br from-gold/10 to-transparent p-4 text-center">
   394→        <Gift className="mx-auto size-10 text-gold" />
   395→        <h4 className="mt-2 text-base font-extrabold text-gold">Earn ₹2,000 per friend!</h4>
   396→        <p className="mt-1 text-xs text-muted-foreground">
   397→          Get ₹2,000 when your friend makes their first deposit. They get ₹1,000 too!
   398→        </p>
   399→      </div>
   400→      <div>
   401→        <p className="mb-1 text-[10px] font-bold tracking-wide text-muted-foreground">YOUR REFERRAL CODE</p>
   402→        <div className="flex items-center gap-2 rounded-xl border border-border bg-card px-3 py-2.5">
   403→          <span className="flex-1 font-mono text-sm font-bold tracking-wider text-bull">{referralCode}</span>
   404→          <button
   405→            onClick={handleCopy}
   406→            className="rounded-md bg-bull/20 px-2.5 py-1.5 text-[10px] font-bold text-bull active:scale-95 transition"
   407→            aria-label="Copy referral code"
   408→          >
   409→            {copied ? <Check className="size-3.5" /> : <Copy className="size-3.5" />}
   410→          </button>
   411→        </div>
   412→      </div>
   413→      <div>
   414→        <p className="mb-1 text-[10px] font-bold tracking-wide text-muted-foreground">SHARE LINK</p>
   415→        <div className="flex items-center gap-2 rounded-xl border border-border bg-card px-3 py-2.5">
   416→          <span className="flex-1 truncate text-xs text-muted-foreground">{referralLink}</span>
   417→          <button
   418→            onClick={handleCopy}
   419→            className="rounded-md bg-info/20 px-2.5 py-1.5 text-[10px] font-bold text-info active:scale-95 transition"
   420→          >
   421→            {copied ? "COPIED" : "COPY"}
   422→          </button>
   423→        </div>
   424→      </div>
   425→      {shareStatus && (
   426→        <p className="text-center text-[10px] font-semibold text-bull">{shareStatus}</p>
   427→      )}
   428→      <div className="grid grid-cols-3 gap-2">
   429→        <ShareBtn label="WhatsApp" emoji="💬" onClick={() => handleShare("whatsapp")} />
   430→        <ShareBtn label="Telegram" emoji="✈️" onClick={() => handleShare("telegram")} />
   431→        <ShareBtn label="More" emoji="⋯" onClick={() => handleShare("more")} />
   432→      </div>
   433→      <div className="rounded-xl border border-border bg-card p-3 text-center">
   434→        <p className="text-[10px] text-muted-foreground">FRIENDS JOINED</p>
   435→        <p className="text-2xl font-extrabold text-bull">0</p>
   436→        <p className="text-[10px] text-muted-foreground">Total earned: ₹0</p>
   437→      </div>
   438→    </div>
   439→  )

   505→function SecurityContent({
   506→  authMethod,
   507→  email,
   508→  phone,
   509→}: {
   510→  authMethod: string | null
   511→  email: string | null
   512→  phone: string | null
   513→}) {
   514→  const authLabel =
   515→    authMethod === "google" ? "Google Account" :
   516→    authMethod === "phone" ? "Phone Number" :
   517→    authMethod === "email" ? "Email & Password" :
   518→    "Guest"
   519→  const [notice, setNotice] = useState<string | null>(null)
   520→  const [twoFAEnabled, setTwoFAEnabled] = useState(false)
   521→
   522→  const flashNotice = (msg: string) => {
   523→    setNotice(msg)
   524→    setTimeout(() => setNotice(null), 2500)
   525→  }
   526→
   527→  const handleChangePassword = () => {
   528→    if (authMethod === "google" || authMethod === "phone") {
   529→      flashNotice(`Password not used — you sign in via ${authLabel}.`)
   530→    } else if (email) {
   531→      flashNotice(`Password reset link sent to ${email}`)
   532→    } else {
   533→      flashNotice("Add an email to your account first to change password.")
   534→    }
   535→  }
   536→
   537→  const handleToggle2FA = () => {
   538→    if (twoFAEnabled) {
   539→      setTwoFAEnabled(false)
   540→      flashNotice("Two-Factor Authentication disabled.")
   541→    } else {
   542→      setTwoFAEnabled(true)
   543→      flashNotice("Two-Factor Authentication enabled — extra security active!")
   544→    }
   545→  }
   546→
   547→  return (
   548→    <div className="space-y-3">
   549→      <div className="rounded-xl border border-bull/30 bg-bull/5 p-3">
   550→        <div className="flex items-center gap-2">
   551→          <Shield className="size-5 text-bull" />
   552→          <p className="text-sm font-bold text-bull">Account Verified</p>
   553→        </div>
   554→        <p className="mt-1 text-[10px] text-muted-foreground">
   555→          Your account is verified via {authLabel}.
   556→        </p>
   557→      </div>
   558→
   559→      <div className="rounded-xl border border-border bg-card p-3">
   560→        <p className="mb-2 text-[10px] font-bold tracking-wide text-muted-foreground">ACCOUNT DETAILS</p>
   561→        <DetailRow icon={<Mail className="size-4 text-muted-foreground" />} label="Email" value={email || "Not set"} />
   562→        <DetailRow icon={<Phone className="size-4 text-muted-foreground" />} label="Phone" value={phone || "Not set"} />
   563→        <DetailRow icon={<Lock className="size-4 text-muted-foreground" />} label="Sign-in method" value={authLabel} />

   566→      <div className="rounded-xl border border-gold/30 bg-gold/5 p-3">
   567→        <div className="flex items-center gap-2">
   568→          <Lock className="size-5 text-gold" />
   569→          <p className="text-sm font-bold text-gold">KYC Status: Verified</p>
   570→        </div>
   571→        <p className="mt-1 text-[10px] text-muted-foreground">
   572→          Your identity has been verified. Withdrawal limit: ₹50,000/day.
   573→        </p>
   574→      </div>
   575→
   576→      {notice && (
   577→        <motion.div
   578→          initial={{ opacity: 0, y: -5 }}
   579→          animate={{ opacity: 1, y: 0 }}
   580→          exit={{ opacity: 0 }}
   581→          className="rounded-lg border border-info/40 bg-info/10 px-3 py-2 text-[11px] font-semibold text-info"
   582→        >
   583→          {notice}
   584→        </motion.div>
   585→      )}
   586→
   587→      <motion.button
   588→        onClick={handleChangePassword}
   589→        whileTap={{ scale: 0.97 }}
   590→        className="flex w-full items-center justify-center gap-2 rounded-xl border border-border bg-card py-3 text-xs font-bold text-muted-foreground hover:border-bull/40 hover:text-bull transition"
   591→      >
   592→        <Shield className="size-4" /> Change Password
   593→      </motion.button>
   594→      <motion.button
   595→        onClick={handleToggle2FA}
   596→        whileTap={{ scale: 0.97 }}
   597→        className={`flex w-full items-center justify-center gap-2 rounded-xl border py-3 text-xs font-bold transition ${
   598→          twoFAEnabled
   599→            ? "border-bull/40 bg-bull/10 text-bull"
   600→            : "border-border bg-card text-muted-foreground hover:border-gold/40 hover:text-gold"
   601→        }`}
   602→      >
   603→        <Lock className="size-4" />
   604→        {twoFAEnabled ? "Two-Factor Authentication: ON" : "Enable Two-Factor Authentication"}
   605→      </motion.button>
   606→    </div>
   607→  )
