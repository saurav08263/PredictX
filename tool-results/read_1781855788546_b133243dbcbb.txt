   137→type Store = {
   138→  // User
   139→  userId: string | null
   140→  userName: string
   141→  email: string | null
   142→  phone: string | null
   143→  country: Country
   144→  setCountry: (c: Country) => void

   152→

   154→  twoFactorEnabled: boolean

   157→  signup: (name: string, email: string, password: string, referralCode?: string) => Promise<{ success: boolean; error?: string }>

   160→  verifyOtp: (phone: string, code: string, referralCode?: string) => Promise<{ success: boolean; error?: string }>
   161→  logout: () => Promise<void>
   162→
   163→  // Prices (from WebSocket or REST fallback)
   164→  prices: Record<Coin, number>
   165→  priceHistory: Record<Coin, number[]>
   166→  mounted: boolean
   167→  wsConnected: boolean
   168→
   169→  // Round info (from WebSocket or REST)
   170→  currentRound: RoundInfo | null
   171→
   172→  // User financials (from API)
   173→  balance: number
   174→  winnings: number
   175→  profit: number
   176→  bonus: number
   177→  wins: number
   178→  totalBets: number
   179→  winRate: number
   180→
   181→  // Active bet (from API)
   182→  activeBet: ActiveBet
   183→
   184→  // Last round result (from WebSocket round:end or REST polling)
   185→  lastResult: RoundResult
   186→
   187→  // Bet history (from API)
   188→  recentBets: BetHistory[]
   189→
   190→  // Transactions (from API)
   191→  transactions: Transaction[]
   192→
   193→  // Actions
   194→  placeBet: (direction: "UP" | "DOWN", coin: Coin, duration: Duration, amount: number) => Promise<{ success: boolean; error?: string; code?: string }>
   195→  deposit: (n: number, method?: string) => Promise<void>
   196→  withdraw: (n: number, method?: string) => Promise<void>
   197→  refreshUser: () => Promise<void>
   198→  refreshBets: () => Promise<void>
   199→  refreshTransactions: () => Promise<void>
   200→  fetchRoundInfo: (coin: Coin, duration: Duration) => Promise<RoundInfo | null>
   201→
   202→  // Win celebration
   203→  winEvent: WinEvent
   204→  clearWinEvent: () => void
   205→
   206→  // Clear last result
   207→  clearLastResult: () => void
   208→
   209→  // Toasts
   210→  toasts: Toast[]
   211→}

   233→  const [authLoading, setAuthLoading] = useState(false)
   234→  const [authError, setAuthError] = useState<string | null>(null)
   235→
   236→  // User
   237→  const [userId, setUserId] = useState<string | null>(null)
   238→  const [userName, setUserName] = useState("Trader")

   241→  const [twoFactorEnabled, setTwoFactorEnabled] = useState(false)
   242→  const [balance, setBalance] = useState(0)
   243→  const [winnings, setWinnings] = useState(0)
   244→  const [profit, setProfit] = useState(0)
   245→  const [bonus, setBonus] = useState(0)
   246→  const [wins, setWins] = useState(0)

   250→  // Prices from WebSocket or REST
   251→  const [prices, setPrices] = useState<Record<Coin, number>>({
   252→    BTCUSDT: 0,

   254→    SOLUSDT: 0,
   255→  })
   256→  const [priceHistory, setPriceHistory] = useState<Record<Coin, number[]>>({
   257→    BTCUSDT: [],
   258→    ETHUSDT: [],
   259→    SOLUSDT: [],
   260→  })
   261→  const [currentRound, setCurrentRound] = useState<RoundInfo | null>(null)
   262→  const [wsConnected, setWsConnected] = useState(false)
   263→
   264→  // Bet state
   265→  const [activeBet, setActiveBet] = useState<ActiveBet>(null)
   266→  const [recentBets, setRecentBets] = useState<BetHistory[]>([])
   267→  const [transactions, setTransactions] = useState<Transaction[]>([])
   268→
   269→  // Round result (entry vs exit price comparison)
   270→  const [lastResult, setLastResult] = useState<RoundResult>(null)
   271→  const clearLastResult = useCallback(() => setLastResult(null), [])
   272→
   273→  // Win event

   277→  // Toasts

   281→    setToasts((t) => [...t, { id, msg, tone }])

   284→
   285→  // Refs to avoid stale closures
   286→  const socketRef = useRef<Socket | null>(null)
   287→  const roundSubRef = useRef<string>("")
   288→  const activeBetRef = useRef<ActiveBet>(null)
   289→  const userIdRef = useRef<string | null>(null)
   290→  const countryRef = useRef<Country>(country)
   291→  const wsConnectedRef = useRef(false)
   292→  const currentRoundRef = useRef<RoundInfo | null>(null)
   293→
   294→  // Keep refs in sync
   295→  useEffect(() => { activeBetRef.current = activeBet }, [activeBet])
   296→  useEffect(() => { userIdRef.current = userId }, [userId])
   297→  useEffect(() => { countryRef.current = country }, [country])

   299→  useEffect(() => { currentRoundRef.current = currentRound }, [currentRound])

   301→  // --- Helper: Update price history from any source ---
   302→  const updatePriceHistory = useCallback((newPrices: Record<Coin, number>) => {
   303→    setPriceHistory((prev) => {
   304→      const updated = { ...prev }
   305→      for (const coin of ["BTCUSDT", "ETHUSDT", "SOLUSDT"] as Coin[]) {
   306→        if (newPrices[coin] > 0) {
   307→          const arr = [...(prev[coin] || []), newPrices[coin]]
   308→          if (arr.length > MAX_HISTORY) arr.shift()
   309→          updated[coin] = arr
   310→        }
   311→      }
   312→      return updated
   313→    })
   314→  }, [])

   317→  useEffect(() => {

   321→        if (r.status === 401) {

   323→          setAuthStatus("unauthenticated")
   324→          return null
   325→        }
   326→        if (!r.ok) {
   327→          // Server error — fall back to login screen rather than staying stuck on loader
   328→          setAuthStatus("unauthenticated")
   329→          return null
   330→        }

   340→        setWinnings(data.winnings)
   341→        setProfit(data.profit)
   342→        setBonus(data.bonus)
   343→        setWins(data.wins)
   344→        setTotalBets(data.totalBets)
   345→        setWinRate(data.winRate)
   346→        setAuthMethod(data.authMethod || null)
   347→        setTwoFactorEnabled(Boolean(data.twoFactorEnabled))
   348→        setAuthStatus("authenticated")
   349→      })
   350→      .catch((e) => {
   351→        console.error(e)
   352→        setAuthStatus("unauthenticated")
   353→      })
   354→  }, [mounted])
   355→
   356→  // --- Auth methods ---
   357→  // Internal: reload user data after auth change (must be defined BEFORE auth methods that use it)

   359→    try {
   360→      const res = await fetch("/api/user")
   361→      if (!res.ok) return
   362→      const data = await res.json()
   363→      setUserId(data.id)
   364→      setUserName(data.name)
   365→      setEmail(data.email || null)
   366→      setPhone(data.phone || null)
   367→      setBalance(data.balance)
   368→      setWinnings(data.winnings)
   369→      setProfit(data.profit)
   370→      setBonus(data.bonus)
   371→      setWins(data.wins)
   372→      setTotalBets(data.totalBets)
   373→      setWinRate(data.winRate)
   374→      setAuthMethod(data.authMethod || null)
   375→      setTwoFactorEnabled(Boolean(data.twoFactorEnabled))
   376→    } catch (e) {
   377→      console.error(e)
   378→    }
   379→  }, [])
   380→
   381→  const login = useCallback(async (email: string, password: string) => {
   382→    setAuthLoading(true)
   383→    setAuthError(null)
   384→    try {
   385→      const res = await fetch("/api/auth/login", {
   386→        method: "POST",
   387→        headers: { "Content-Type": "application/json" },
   388→        body: JSON.stringify({ email, password }),
   389→      })
   390→      const data = await res.json()
   391→      if (!res.ok) {
   392→        setAuthError(data.error || "Login failed")
   393→        return { success: false, error: data.error }
   394→      }
   395→      // Reload full user data
   396→      await refreshUserInternal()
   397→      setAuthMethod(data.authMethod || "email")
   398→      setAuthStatus("authenticated")
   399→      return { success: true }
   400→    } catch {
   401→      setAuthError("Network error — please try again")
   402→      return { success: false, error: "Network error" }
   403→    } finally {
   404→      setAuthLoading(false)
   405→    }
   406→  }, [refreshUserInternal])
   407→

   409→    setAuthLoading(true)
   410→    setAuthError(null)
   411→    try {
   412→      const res = await fetch("/api/auth/signup", {
   413→        method: "POST",

   423→      setAuthMethod(data.authMethod || "email")
   424→      setAuthStatus("authenticated")
   425→      return { success: true }
   426→    } catch {
   427→      setAuthError("Network error — please try again")
   428→      return { success: false, error: "Network error" }
   429→    } finally {
   430→      setAuthLoading(false)
   431→    }
   432→  }, [refreshUserInternal])

   542→  // --- Fetch round info via REST API (works without WebSocket) ---

   546→      if (res.ok) {

   548→        const info: RoundInfo = {
   549→          roundId: data.roundId,
   550→          coin: data.coin as Coin,
   551→          duration: data.duration,
   552→          timeLeft: data.timeLeft,
   553→          entryPrice: data.entryPrice,
   554→          currentPrice: data.currentPrice || data.entryPrice,
   555→        }
   556→        // Update current round state
   557→        setCurrentRound(info)
   558→        return info
   559→      }
   560→    } catch {
   561→      // Silently fail
   562→    }

   582→            data.activeBet.round.coin as Coin,
   583→            data.activeBet.round.duration as Duration,
   584→          )
   585→        }
   586→      }
   587→    } catch (err) {
   588→      console.error("Failed to refresh active bet:", err)
   589→    }
   590→  }, [subscribeToRound, fetchRoundInfo])

   592→  // --- Refresh user data ---

   603→        setBonus(data.bonus)

  1110→export function timeLabels(count = 6) {
  1111→  const now = new Date()
  1112→  const labels: string[] = []
  1113→  for (let i = count - 1; i >= 0; i--) {
  1114→    const d = new Date(now.getTime() - i * 60_000)
  1115→    labels.push(`${String(d.getHours()).padStart(2, "0")}:${String(d.getMinutes()).padStart(2, "0")}`)
  1116→  }
  1117→  return labels
  1118→}

  1120→export function priceLabels(history: number[], count = 7) {
  1121→  if (history.length === 0) return Array.from({ length: count }, () => "--.--")
  1122→  const min = Math.min(...history)
  1123→  const max = Math.max(...history)
  1124→  const pad = (max - min) * 0.1 || 100
  1125→  const lo = min - pad
  1126→  const hi = max + pad
  1127→  const labels: string[] = []
  1128→  for (let i = 0; i < count; i++) {
  1129→    const v = hi - ((hi - lo) / (count - 1)) * i
  1130→    labels.push(v.toLocaleString("en-US", { minimumFractionDigits: 2, maximumFractionDigits: 2 }))
  1131→  }
  1132→  return labels
  1133→}
