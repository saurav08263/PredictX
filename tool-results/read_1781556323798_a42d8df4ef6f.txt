     1→"use client"
     2→
     3→import { useState, useEffect, useCallback } from "react"
     4→import { motion, AnimatePresence } from "framer-motion"
     5→import {
     6→  ChevronUp,
     7→  ChevronDown,
     8→  ArrowUpRight,
     9→  ArrowDownRight,
    10→  Timer,
    11→  Check,
    12→  Zap,
    13→  Plus,
    14→  TrendingUp,
    15→  Clock,
    16→} from "lucide-react"
    17→import { PriceChart } from "../price-chart"
    18→import {
    19→  useCrypto,
    20→  formatClock,
    21→  formatPrice,
    22→  COINS,
    23→  DURATIONS,
    24→  AMOUNT_OPTIONS,
    25→  type Coin,
    26→  type Duration,
    27→} from "../store"
    28→import type { Tab } from "../app-shell"
    29→
    30→type Step = "coin" | "time" | "predict" | "amount" | "confirm" | "live" | "result"
    31→
    32→export function PredictScreen({ onNavigate }: { onNavigate: (t: Tab) => void }) {
    33→  const {
    34→    country,
    35→    fmt,
    36→    mounted,
    37→    prices,
    38→    priceHistory,
    39→    currentRound,
    40→    wsConnected,
    41→    balance,
    42→    activeBet,
    43→    lastResult,
    44→    placeBet,
    45→    subscribeToRound,
    46→    fetchRoundInfo,
    47→    winEvent,
    48→    clearWinEvent,
    49→    clearLastResult,
    50→    tier,
    51→    dailyPredictions,
    52→    dailyLimit,
    53→    activatePremium,
    54→  } = useCrypto()
    55→
    56→  const [step, setStep] = useState<Step>("coin")
    57→  const [selectedCoin, setSelectedCoin] = useState<Coin>("BTCUSDT")
    58→  const [selectedDuration, setSelectedDuration] = useState<Duration>(60)
    59→  const [direction, setDirection] = useState<"UP" | "DOWN" | null>(null)
    60→  const [selectedAmount, setSelectedAmount] = useState<number>(100)
    61→  const [customAmount, setCustomAmount] = useState("")
    62→  const [betPlacing, setBetPlacing] = useState(false)
    63→  const [showPremiumModal, setShowPremiumModal] = useState(false)
    64→
    65→  // Derive step from store state reactively
    66→  const effectiveStep: Step = lastResult
    67→    ? "result"
    68→    : (activeBet && step !== "result" ? "live" : step)
    69→
    70→  // When there's an active bet, derive coin/duration from it
    71→  const displayCoin: Coin = activeBet?.round?.coin ? (activeBet.round.coin as Coin) : selectedCoin
    72→  const displayDuration: Duration = activeBet?.round?.duration ? (activeBet.round.duration as Duration) : selectedDuration
    73→
    74→  const coinData = COINS.find((c) => c.id === displayCoin)!
    75→  const currentPrice = prices[displayCoin]
    76→  const history = priceHistory[displayCoin] || []
    77→
    78→  // Subscribe to round when coin/duration changes
    79→  useEffect(() => {
    80→    if (!mounted) return
    81→    // Always subscribe via WS if available
    82→    subscribeToRound(selectedCoin, selectedDuration)
    83→    // Also fetch round info via REST (works even without WS)
    84→    fetchRoundInfo(selectedCoin, selectedDuration)
    85→  }, [mounted, selectedCoin, selectedDuration, subscribeToRound, fetchRoundInfo])
    86→
    87→  // Handle coin selection
    88→  const handleSelectCoin = (coin: Coin) => {
    89→    setSelectedCoin(coin)
    90→    setStep("time")
    91→  }
    92→
    93→  // Handle time selection
    94→  const handleSelectTime = (d: Duration) => {
    95→    setSelectedDuration(d)
    96→    setStep("predict")
    97→  }
    98→
    99→  // Handle UP/DOWN selection
   100→  const handleSelectDirection = (dir: "UP" | "DOWN") => {
   101→    setDirection(dir)
   102→    setStep("amount")
   103→  }
   104→
   105→  // Get the effective bet amount
   106→  const betAmount = customAmount ? Number(customAmount) : selectedAmount
   107→
   108→  // Confirm bet
   109→  const handleConfirmBet = async () => {
   110→    if (!direction || betAmount <= 0) return
   111→
   112→    // Check daily limit for free users
   113→    if (tier === "FREE" && dailyLimit > 0 && dailyPredictions >= dailyLimit) {
   114→      setShowPremiumModal(true)
   115→      return
   116→    }
   117→
   118→    setBetPlacing(true)
   119→    try {
   120→      const result = await placeBet(direction, selectedCoin, selectedDuration, betAmount)
   121→      setBetPlacing(false)
   122→      if (result.success) {
   123→        setStep("live")
   124→      } else if (result.code === "DAILY_LIMIT_REACHED") {
   125→        setShowPremiumModal(true)
   126→      }
   127→    } catch {
   128→      setBetPlacing(false)
   129→    }
   130→  }
   131→
   132→  // Play again
   133→  const handlePlayAgain = () => {
   134→    setDirection(null)
   135→    setCustomAmount("")
   136→    setSelectedAmount(100)
   137→    clearWinEvent()
   138→    clearLastResult()
   139→    setStep("coin")
   140→  }
   141→
   142→  // Back navigation
   143→  const goBack = () => {
   144→    switch (step) {
   145→      case "time": setStep("coin"); break
   146→      case "predict": setStep("time"); break
   147→      case "amount": setStep("predict"); break
   148→      case "confirm": setStep("amount"); break
   149→      default: setStep("coin"); break
   150→    }
   151→  }
   152→
   153→  // Get price change color class
   154→  const getPriceChangeColor = (entry: number, exit: number) => {
   155→    if (exit > entry) return "text-bull"
   156→    if (exit < entry) return "text-bear"
   157→    return "text-muted-foreground"
   158→  }
   159→
   160→  return (
   161→    <div className="flex flex-col gap-3 px-4 pb-4 pt-3">
   162→      {/* Daily limit indicator for free users */}
   163→      {tier === "FREE" && (
   164→        <div className="flex items-center justify-between rounded-lg border border-gold/30 bg-gold/5 px-3 py-2">
   165→          <div className="flex items-center gap-2">
   166→            <Zap className="size-3.5 text-gold" />
   167→            <span className="text-[10px] font-bold text-gold">
   168→              {dailyLimit > 0 ? `${Math.max(0, dailyLimit - dailyPredictions)}/${dailyLimit}` : "0/3"} FREE TODAY
   169→            </span>
   170→          </div>
   171→          <button
   172→            onClick={() => setShowPremiumModal(true)}
   173→            className="rounded-md bg-gold/20 px-2 py-0.5 text-[9px] font-bold text-gold hover:bg-gold/30 transition"
   174→          >
   175→            UNLIMITED →
   176→          </button>
   177→        </div>
   178→      )}
   179→
   180→      {/* Premium badge for premium users */}
   181→      {tier === "PREMIUM" && (
   182→        <div className="flex items-center justify-center gap-1.5 rounded-lg border border-gold/40 bg-gold/10 px-3 py-1.5">
   183→          <Zap className="size-3 text-gold" />
   184→          <span className="text-[10px] font-bold text-gold">PREMIUM · UNLIMITED PREDICTIONS</span>
   185→        </div>
   186→      )}
   187→
   188→      {/* Step indicator */}
   189→      <div className="flex items-center gap-1.5">
   190→        {(["coin", "time", "predict", "amount", "confirm"] as Step[]).map((s, i) => (
   191→          <div
   192→            key={s}
   193→            className={`h-1.5 flex-1 rounded-full transition-all duration-300 ${
   194→              effectiveStep === s || (effectiveStep === "live" && i <= 4) || (effectiveStep === "result" && i <= 4)
   195→                ? "bg-bull"
   196→                : ["coin", "time", "predict", "amount", "confirm"].indexOf(effectiveStep) > i
   197→                  ? "bg-bull/40"
   198→                  : "bg-border"
   199→            }`}
   200→          />
   201→        ))}
   202→      </div>
   203→
   204→      <AnimatePresence mode="wait">
   205→        {/* ====== STEP 1: SELECT COIN ====== */}
   206→        {effectiveStep === "coin" && (
   207→          <motion.div
   208→            key="coin"
   209→            initial={{ opacity: 0, x: 20 }}
   210→            animate={{ opacity: 1, x: 0 }}
   211→            exit={{ opacity: 0, x: -20 }}
   212→            className="flex flex-col gap-4"
   213→          >
   214→            <div className="text-center">
   215→              <p className="text-[10px] font-semibold tracking-widest text-muted-foreground">STEP 1 OF 5</p>
   216→              <h2 className="text-xl font-extrabold mt-1">Select Coin</h2>
   217→            </div>
   218→
   219→            <div className="flex flex-col gap-3">
   220→              {COINS.map((coin) => {
   221→                const price = prices[coin.id]
   222→                const coinHistory = priceHistory[coin.id] || []
   223→                // Only compute change if we have enough history and a valid base price
   224→                const firstValidPrice = coinHistory.find((p) => p > 0)
   225→                const change = firstValidPrice && firstValidPrice > 0 && price > 0
   226→                  ? ((price - firstValidPrice) / firstValidPrice * 100)
   227→                  : 0
   228→                const isUp = change >= 0
   229→
   230→                return (
   231→                  <motion.button
   232→                    key={coin.id}
   233→                    onClick={() => handleSelectCoin(coin.id)}
   234→                    className={`flex items-center gap-4 rounded-xl border p-4 text-left transition-all ${
   235→                      selectedCoin === coin.id
   236→                        ? "border-bull/50 bg-bull/10"
   237→                        : "border-border bg-card hover:border-bull/30"
   238→                    }`}
   239→                    whileTap={{ scale: 0.97 }}
   240→                  >
   241→                    <span className={`flex size-12 items-center justify-center rounded-full text-xl font-bold ${
   242→                      coin.id === "BTCUSDT" ? "bg-gold/20 text-gold" :
   243→                      coin.id === "ETHUSDT" ? "bg-blue-500/20 text-blue-400" :
   244→                      "bg-purple-500/20 text-purple-400"
   245→                    }`}>
   246→                      {coin.icon}
   247→                    </span>
   248→                    <div className="flex-1">
   249→                      <p className="font-bold">{coin.label}</p>
   250→                      <p className="text-xs text-muted-foreground">USDT Pair</p>
   251→                    </div>
   252→                    <div className="text-right">
   253→                      <p className="font-extrabold tabular-nums">
   254→                        {mounted && price > 0 ? `$${formatPrice(price, coin.id)}` : "--.--"}
   255→                      </p>
   256→                      <p className={`text-xs font-semibold flex items-center justify-end gap-0.5 ${isUp ? "text-bull" : "text-bear"}`}>
   257→                        {mounted && price > 0 ? (
   258→                          <>
   259→                            {isUp ? <ChevronUp className="size-3" /> : <ChevronDown className="size-3" />}
   260→                            {Math.abs(change).toFixed(2)}%
   261→                          </>
   262→                        ) : "--.--%"}
   263→                      </p>
   264→                    </div>
   265→                  </motion.button>
   266→                )
   267→              })}
   268→            </div>
   269→
   270→            {wsConnected && (
   271→              <p className="flex items-center justify-center gap-1.5 text-[10px] text-bull">
   272→                <span className="size-1.5 rounded-full bg-bull animate-pulse" /> LIVE PRICES FROM BINANCE
   273→              </p>
   274→            )}
   275→          </motion.div>
   276→        )}
   277→
   278→        {/* ====== STEP 2: SELECT TIME ====== */}
   279→        {effectiveStep === "time" && (
   280→          <motion.div
   281→            key="time"
   282→            initial={{ opacity: 0, x: 20 }}
   283→            animate={{ opacity: 1, x: 0 }}
   284→            exit={{ opacity: 0, x: -20 }}
   285→            className="flex flex-col gap-4"
   286→          >
   287→            <button onClick={goBack} className="text-xs text-muted-foreground hover:text-foreground transition">← Back</button>
   288→            <div className="text-center">
   289→              <p className="text-[10px] font-semibold tracking-widest text-muted-foreground">STEP 2 OF 5</p>
   290→              <h2 className="text-xl font-extrabold mt-1">Select Time</h2>
   291→              <p className="text-xs text-muted-foreground mt-1">How long until the round ends?</p>
   292→            </div>
   293→
   294→            <div className="grid grid-cols-4 gap-2.5">
   295→              {DURATIONS.map((d) => (
   296→                <motion.button
   297→                  key={d.value}
   298→                  onClick={() => handleSelectTime(d.value)}
   299→                  className={`flex flex-col items-center gap-1.5 rounded-xl border py-4 transition-all ${
   300→                    selectedDuration === d.value
   301→                      ? "border-bull/50 bg-bull/10 text-bull"
   302→                      : "border-border bg-card hover:border-bull/30"
   303→                  }`}
   304→                  whileTap={{ scale: 0.95 }}
   305→                >
   306→                  <Clock className="size-5" />
   307→                  <span className="text-lg font-extrabold">{d.label}</span>
   308→                  <span className="text-[9px] font-semibold text-muted-foreground">{d.value}s ROUND</span>
   309→                </motion.button>
   310→              ))}
   311→            </div>
   312→
   313→            {/* Current price preview */}
   314→            <div className="rounded-xl border border-border bg-card p-3 text-center">
   315→              <p className="text-[10px] font-semibold text-muted-foreground">{coinData.label} CURRENT PRICE</p>
   316→              <p className="text-2xl font-extrabold tabular-nums mt-1">
   317→                {mounted && currentPrice > 0 ? `$${formatPrice(currentPrice, selectedCoin)}` : "--.--"}
   318→              </p>
   319→            </div>
   320→          </motion.div>
   321→        )}
   322→
   323→        {/* ====== STEP 3 & 4: PRICE DISPLAY + UP/DOWN ====== */}
   324→        {effectiveStep === "predict" && (
   325→          <motion.div
   326→            key="predict"
   327→            initial={{ opacity: 0, x: 20 }}
   328→            animate={{ opacity: 1, x: 0 }}
   329→            exit={{ opacity: 0, x: -20 }}
   330→            className="flex flex-col gap-3"
   331→          >
   332→            <button onClick={goBack} className="text-xs text-muted-foreground hover:text-foreground transition">← Back</button>
   333→
   334→            {/* Coin + Duration header */}
   335→            <div className="flex items-center justify-between">
   336→              <div className="flex items-center gap-2">
   337→                <span className={`flex size-8 items-center justify-center rounded-full text-sm font-bold ${
   338→                  selectedCoin === "BTCUSDT" ? "bg-gold/20 text-gold" :
   339→                  selectedCoin === "ETHUSDT" ? "bg-blue-500/20 text-blue-400" :
   340→                  "bg-purple-500/20 text-purple-400"
   341→                }`}>
   342→                  {coinData.icon}
   343→                </span>
   344→                <div>
   345→                  <p className="font-bold">{coinData.label}</p>
   346→                  <p className="text-[10px] text-muted-foreground">{selectedDuration}s round</p>
   347→                </div>
   348→              </div>
   349→              <span className="flex items-center gap-1 rounded-md border border-bull/40 bg-bull/10 px-2 py-1 text-xs font-bold text-bull">
   350→                <span className="size-1.5 rounded-full bg-bull animate-pulse" /> LIVE
   351→              </span>
   352→            </div>
   353→
   354→            {/* Big price display */}
   355→            <div className="rounded-2xl border border-border bg-card p-5 text-center">
   356→              <p className="text-[10px] font-semibold tracking-wide text-muted-foreground">CURRENT PRICE</p>
   357→              <p className="text-4xl font-extrabold tabular-nums mt-2">
   358→                {mounted && currentPrice > 0 ? `$${formatPrice(currentPrice, selectedCoin)}` : "--.--"}
   359→              </p>
   360→              <p className="text-xs text-muted-foreground mt-1">USDT</p>
   361→            </div>
   362→
   363→            {/* Mini chart */}
   364→            {history.length > 2 && (
   365→              <div className="rounded-xl border border-border bg-panel p-2">
   366→                <PriceChart
   367→                  points={history.slice(-40)}
   368→                  className="h-28 w-full"
   369→                  entryPrice={null}
   370→                  direction={null}
   371→                />
   372→              </div>
   373→            )}
   374→
   375→            {/* Round timer */}
   376→            {currentRound && (
   377→              <div className="flex items-center justify-between rounded-xl border border-bull/40 bg-bull/5 px-3 py-2">
   378→                <div className="flex items-center gap-2">
   379→                  <Timer className="size-4 text-bull" />
   380→                  <span className="text-xs font-bold">ROUND ENDS IN</span>
   381→                </div>
   382→                <span className="text-lg font-extrabold tabular-nums text-bull">
   383→                  {formatClock(currentRound.timeLeft)}
   384→                </span>
   385→              </div>
   386→            )}
   387→
   388→            {/* Step 3 label */}
   389→            <div className="text-center">
   390→              <p className="text-[10px] font-semibold tracking-widest text-muted-foreground">STEP 3 & 4</p>
   391→              <h2 className="text-lg font-extrabold mt-0.5">Predict the Price</h2>
   392→              <p className="text-xs text-muted-foreground">Will it go UP or DOWN?</p>
   393→            </div>
   394→
   395→            {/* UP/DOWN buttons */}
   396→            <div className="grid grid-cols-2 gap-3">
   397→              <motion.button
   398→                onClick={() => handleSelectDirection("UP")}
   399→                className="flex flex-col items-center gap-2 rounded-2xl border-2 border-bull/30 bg-bull/10 py-6 transition-all hover:border-bull/60 hover:bg-bull/20"
   400→                whileTap={{ scale: 0.95 }}
   401→              >
   402→                <ArrowUpRight className="size-10 text-bull" />
   403→                <span className="text-xl font-extrabold text-bull">UP</span>
   404→                <span className="text-[10px] font-semibold text-bull/70">Price will rise</span>
   405→              </motion.button>
   406→              <motion.button
   407→                onClick={() => handleSelectDirection("DOWN")}
   408→                className="flex flex-col items-center gap-2 rounded-2xl border-2 border-bear/30 bg-bear/10 py-6 transition-all hover:border-bear/60 hover:bg-bear/20"
   409→                whileTap={{ scale: 0.95 }}
   410→              >
   411→                <ArrowDownRight className="size-10 text-bear" />
   412→                <span className="text-xl font-extrabold text-bear">DOWN</span>
   413→                <span className="text-[10px] font-semibold text-bear/70">Price will fall</span>
   414→              </motion.button>
   415→            </div>
   416→          </motion.div>
   417→        )}
   418→
   419→        {/* ====== STEP 5: SELECT AMOUNT ====== */}
   420→        {effectiveStep === "amount" && (
   421→          <motion.div
   422→            key="amount"
   423→            initial={{ opacity: 0, x: 20 }}
   424→            animate={{ opacity: 1, x: 0 }}
   425→            exit={{ opacity: 0, x: -20 }}
   426→            className="flex flex-col gap-4"
   427→          >
   428→            <button onClick={goBack} className="text-xs text-muted-foreground hover:text-foreground transition">← Back</button>
   429→
   430→            <div className="text-center">
   431→              <p className="text-[10px] font-semibold tracking-widest text-muted-foreground">STEP 5 OF 5</p>
   432→              <h2 className="text-xl font-extrabold mt-1">Enter Amount</h2>
   433→            </div>
   434→
   435→            {/* Direction summary */}
   436→            <div className={`flex items-center justify-center gap-2 rounded-xl border py-3 ${
   437→              direction === "UP" ? "border-bull/40 bg-bull/10 text-bull" : "border-bear/40 bg-bear/10 text-bear"
   438→            }`}>
   439→              {direction === "UP" ? <ArrowUpRight className="size-5" /> : <ArrowDownRight className="size-5" />}
   440→              <span className="font-extrabold">{direction} · {coinData.label}</span>
   441→              <span className="text-xs opacity-70">· {selectedDuration}s</span>
   442→            </div>
   443→
   444→            {/* Quick amount buttons */}
   445→            <div className="flex items-center gap-2">
   446→              {AMOUNT_OPTIONS.map((a) => (
   447→                <motion.button
   448→                  key={a}
   449→                  onClick={() => { setSelectedAmount(a); setCustomAmount("") }}
   450→                  className={`flex-1 rounded-lg border py-3 text-sm font-bold transition ${
   451→                    selectedAmount === a && !customAmount
   452→                      ? "border-bull bg-bull/15 text-bull"
   453→                      : "border-border bg-card text-foreground"
   454→                  }`}
   455→                  whileTap={{ scale: 0.95 }}
   456→                >
   457→                  {country.symbol.trim()}{a.toLocaleString(country.locale)}
   458→                </motion.button>
   459→              ))}
   460→            </div>
   461→
   462→            {/* Custom amount input */}
   463→            <div>
   464→              <p className="text-[10px] font-semibold tracking-wide text-muted-foreground mb-2">CUSTOM AMOUNT</p>
   465→              <div className="flex items-center rounded-xl border border-border bg-card px-4 py-3 focus-within:border-bull/50 transition-colors">
   466→                <span className="text-lg font-bold text-muted-foreground">{country.symbol.trim()}</span>
   467→                <input
   468→                  autoFocus
   469→                  inputMode="numeric"
   470→                  value={customAmount}
   471→                  onChange={(e) => setCustomAmount(e.target.value.replace(/[^0-9]/g, ""))}
   472→                  placeholder="Enter amount"
   473→                  className="w-full bg-transparent px-2 text-xl font-extrabold outline-none"
   474→                />
   475→              </div>
   476→            </div>
   477→
   478→            {/* Balance display */}
   479→            <div className="flex items-center justify-between rounded-lg bg-card border border-border px-3 py-2">
   480→              <span className="text-xs text-muted-foreground">Your Balance</span>
   481→              <span className="text-sm font-bold">{fmt(balance, 2)}</span>
   482→            </div>
   483→
   484→            {/* Confirm button */}
   485→            <motion.button
   486→              onClick={() => setStep("confirm")}
   487→              disabled={betAmount <= 0 || betAmount > balance}
   488→              className="flex items-center justify-center gap-2 rounded-xl bg-bull py-3.5 text-sm font-bold text-black glow-green transition disabled:opacity-40"
   489→              whileTap={{ scale: 0.97 }}
   490→            >
   491→              <Check className="size-4" /> CONTINUE TO CONFIRM
   492→            </motion.button>
   493→          </motion.div>
   494→        )}
   495→
   496→        {/* ====== STEP 6: CONFIRM BET ====== */}
   497→        {effectiveStep === "confirm" && (
   498→          <motion.div
   499→            key="confirm"
   500→            initial={{ opacity: 0, scale: 0.95 }}
   501→            animate={{ opacity: 1, scale: 1 }}
   502→            exit={{ opacity: 0, scale: 0.95 }}
   503→            className="flex flex-col gap-4"
   504→          >
   505→            <button onClick={goBack} className="text-xs text-muted-foreground hover:text-foreground transition">← Back</button>
   506→
   507→            <div className="text-center">
   508→              <h2 className="text-xl font-extrabold">Confirm Bet</h2>
   509→              <p className="text-xs text-muted-foreground">Review your prediction</p>
   510→            </div>
   511→
   512→            {/* Bet summary card */}
   513→            <div className="rounded-2xl border border-border bg-card p-5 flex flex-col gap-4">
   514→              {/* Coin */}
   515→              <div className="flex items-center justify-between">
   516→                <span className="text-xs text-muted-foreground">Coin</span>

   929→function PremiumPlanButton({
   930→  label,
   931→  price,
   932→  plan,
   933→  badge,
   934→  onActivate,
   935→  onClose,
   936→}: {
   937→  label: string
   938→  price: string
   939→  plan: "monthly" | "yearly"
   940→  badge?: string
   941→  onActivate: (plan: "monthly" | "yearly") => Promise<{ success: boolean }>
   942→  onClose: () => void
   943→}) {
   944→  const [loading, setLoading] = useState(false)
   945→
   946→  const handleActivate = async () => {
   947→    setLoading(true)
   948→    const result = await onActivate(plan)
   949→    setLoading(false)
   950→    if (result.success) {
   951→      onClose()
   952→    }
   953→  }
   954→
   955→  return (
   956→    <motion.button
   957→      onClick={handleActivate}
   958→      disabled={loading}
   959→      className={`relative flex w-full items-center justify-between rounded-xl border px-4 py-3.5 text-left transition disabled:opacity-60 ${
   960→        badge ? "border-gold/50 bg-gold/10" : "border-border bg-card"
   961→      }`}
   962→      whileTap={{ scale: 0.97 }}
   963→    >
   964→      {badge && (
   965→        <span className="absolute -top-2 right-3 rounded-md bg-gold px-2 py-0.5 text-[9px] font-bold text-black">
   966→          {badge}
   967→        </span>
   968→      )}
   969→      <div>
   970→        <p className="font-bold">{label}</p>
   971→        <p className="text-xs text-muted-foreground">Cancel anytime</p>
   972→      </div>
   973→      <div className="flex items-center gap-2">
   974→        <span className="text-lg font-extrabold text-bull">{price}</span>
   975→        {loading && <span className="size-4 animate-spin rounded-full border-2 border-gold border-t-transparent" />}
   976→      </div>
   977→    </motion.button>
   978→  )
   979→}
