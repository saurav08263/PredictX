   138→  // User

   167→  premiumExpiresAt: string | null
   168→  activatePremium: (plan: "monthly" | "yearly") => Promise<{ success: boolean; error?: string }>
   169→  cancelPremium: () => Promise<void>
   170→
   171→  // Active bet (from API)
   172→  activeBet: ActiveBet
   173→
   174→  // Last round result (from WebSocket round:end or REST polling)
   175→  lastResult: RoundResult
   176→

   179→
   180→  // Transactions (from API)
   181→  transactions: Transaction[]
   182→

   184→  placeBet: (direction: "UP" | "DOWN", coin: Coin, duration: Duration, amount: number) => Promise<{ success: boolean; error?: string; code?: string }>

   195→
   196→  // Clear last result

   198→
   199→  // Toasts

   217→  const mounted = useHydrated()
   218→  const [country, setCountry] = useState<Country>(COUNTRIES[0])
   219→
   220→  // User
   221→  const [userId, setUserId] = useState<string | null>(null)
   222→  const [userName, setUserName] = useState("Trader")
   223→  const [balance, setBalance] = useState(0)
   224→  const [winnings, setWinnings] = useState(0)
   225→  const [profit, setProfit] = useState(0)
   226→  const [bonus, setBonus] = useState(0)
   227→  const [wins, setWins] = useState(0)
   228→  const [totalBets, setTotalBets] = useState(0)
   229→  const [winRate, setWinRate] = useState(0)
   230→
   231→  // Premium tier
   232→  const [tier, setTier] = useState<"FREE" | "PREMIUM">("FREE")
   233→  const [dailyPredictions, setDailyPredictions] = useState(0)
   234→  const [dailyLimit, setDailyLimit] = useState(3)
   235→  const [premiumExpiresAt, setPremiumExpiresAt] = useState<string | null>(null)
   236→
   237→  // Prices from WebSocket or REST
   238→  const [prices, setPrices] = useState<Record<Coin, number>>({
   239→    BTCUSDT: 0,
   240→    ETHUSDT: 0,
   241→    SOLUSDT: 0,
   242→  })
   243→  const [priceHistory, setPriceHistory] = useState<Record<Coin, number[]>>({
   244→    BTCUSDT: [],
   245→    ETHUSDT: [],
   246→    SOLUSDT: [],
   247→  })
   248→  const [currentRound, setCurrentRound] = useState<RoundInfo | null>(null)
   249→  const [wsConnected, setWsConnected] = useState(false)
   250→
   251→  // Bet state
   252→  const [activeBet, setActiveBet] = useState<ActiveBet>(null)
   253→  const [recentBets, setRecentBets] = useState<BetHistory[]>([])
   254→  const [transactions, setTransactions] = useState<Transaction[]>([])
   255→
   256→  // Round result (entry vs exit price comparison)
   257→  const [lastResult, setLastResult] = useState<RoundResult>(null)
   258→  const clearLastResult = useCallback(() => setLastResult(null), [])
   259→
   260→  // Win event
   261→  const [winEvent, setWinEvent] = useState<WinEvent>(null)
   262→  const clearWinEvent = useCallback(() => setWinEvent(null), [])
   263→
   264→  // Toasts
   265→  const [toasts, setToasts] = useState<Toast[]>([])
   266→  const pushToast = useCallback((msg: string, tone: Toast["tone"]) => {
   267→    const id = Date.now() + Math.random()

   271→
   272→  // Refs to avoid stale closures
   273→  const socketRef = useRef<Socket | null>(null)
   274→  const roundSubRef = useRef<string>("")
   275→  const activeBetRef = useRef<ActiveBet>(null)
   276→  const userIdRef = useRef<string | null>(null)

   278→  const wsConnectedRef = useRef(false)
   279→  const currentRoundRef = useRef<RoundInfo | null>(null)
   280→
   281→  // Keep refs in sync

   294→          const arr = [...(prev[coin] || []), newPrices[coin]]
   295→          if (arr.length > MAX_HISTORY) arr.shift()
   296→          updated[coin] = arr
   297→        }
   298→      }
   299→      return updated

   314→        setBonus(data.bonus)

   317→        setWinRate(data.winRate)

   320→        setDailyLimit(data.dailyLimit ?? 3)
   321→        setPremiumExpiresAt(data.premiumExpiresAt || null)
   322→      })
   323→      .catch(console.error)
   324→  }, [mounted])
   325→
   326→  // --- Subscribe to a specific coin+duration round ---
   327→  const subscribeToRound = useCallback((coin: Coin, duration: Duration) => {
   328→    const key = `${coin}-${duration}`
   329→    roundSubRef.current = key
   330→    socketRef.current?.emit("round:subscribe", { coin, duration })
   331→  }, [])
   332→
   333→  // --- Fetch round info via REST API (works without WebSocket) ---
   334→  const fetchRoundInfo = useCallback(async (coin: Coin, duration: Duration): Promise<RoundInfo | null> => {
   335→    try {
   336→      const res = await fetch(`/api/round/current?coin=${coin}&duration=${duration}`)
   337→      if (res.ok) {
   338→        const data = await res.json()
   339→        const info: RoundInfo = {
   340→          roundId: data.roundId,
   341→          coin: data.coin as Coin,
   342→          duration: data.duration,
   343→          timeLeft: data.timeLeft,
   344→          entryPrice: data.entryPrice,
   345→          currentPrice: data.currentPrice || data.entryPrice,
   346→        }
   347→        // Update current round state
   348→        setCurrentRound(info)
   349→        return info
   350→      }
   351→    } catch {
   352→      // Silently fail
   353→    }
   354→    return null
   355→  }, [])
   356→
   357→  // --- Fetch active bet ---
   358→  const refreshActiveBet = useCallback(async () => {
   359→    const uid = userIdRef.current
   360→    if (!uid) return
   361→    try {
   362→      const res = await fetch(`/api/bet/active?userId=${uid}`)
   363→      if (res.ok) {
   364→        const data = await res.json()
   365→        setActiveBet(data.activeBet)
   366→        if (data.activeBet) {
   367→          subscribeToRound(
   368→            data.activeBet.round.coin as Coin,
   369→            data.activeBet.round.duration as Duration,
   370→          )
   371→          // Also try to fetch round info via REST
   372→          fetchRoundInfo(
   373→            data.activeBet.round.coin as Coin,
   374→            data.activeBet.round.duration as Duration,
   375→          )
   376→        }
   377→      }
   378→    } catch (err) {
   379→      console.error("Failed to refresh active bet:", err)
   380→    }
   381→  }, [subscribeToRound, fetchRoundInfo])
   382→
   383→  // --- Refresh user data ---
   384→  const refreshUser = useCallback(async () => {
   385→    const uid = userIdRef.current
   386→    if (!uid) return
   387→    try {
   388→      const res = await fetch("/api/user")
   389→      if (res.ok) {
   390→        const data = await res.json()
   391→        setBalance(data.balance)
   392→        setWinnings(data.winnings)
   393→        setProfit(data.profit)
   394→        setBonus(data.bonus)
   395→        setWins(data.wins)
   396→        setTotalBets(data.totalBets)
   397→        setWinRate(data.winRate)
   398→        setTier(data.tier || "FREE")
   399→        setDailyPredictions(data.dailyPredictions || 0)
   400→        setDailyLimit(data.dailyLimit ?? 3)
   401→        setPremiumExpiresAt(data.premiumExpiresAt || null)
   402→      }
   403→    } catch (err) {
   404→      console.error("Failed to refresh user:", err)
   405→    }
   406→  }, [])
   407→
   408→  // --- Refresh bet history ---
   409→  const refreshBets = useCallback(async () => {
   410→    const uid = userIdRef.current
   411→    if (!uid) return
   412→    try {
   413→      const res = await fetch(`/api/bets?userId=${uid}`)
   414→      if (res.ok) {
   415→        const data = await res.json()
   416→        setRecentBets(data.bets || [])
   417→      }
   418→    } catch (err) {
   419→      console.error("Failed to refresh bets:", err)
   420→    }
   421→  }, [])
   422→
   423→  // --- Refresh transactions ---
   424→  const refreshTransactions = useCallback(async () => {
   425→    const uid = userIdRef.current
   426→    if (!uid) return
   427→    try {
   428→      const res = await fetch(`/api/transactions?userId=${uid}`)
   429→      if (res.ok) {
   430→        const data = await res.json()
   431→        setTransactions(data.transactions || [])
   432→      }
   433→    } catch (err) {
   434→      console.error("Failed to refresh transactions:", err)
   435→    }
   436→  }, [])
   437→
   438→  // --- REST API price polling fallback ---
   439→  useEffect(() => {
   440→    if (!mounted) return
   441→
   442→    const fetchPrices = async () => {
   443→      try {
   444→        const res = await fetch("/api/prices")
   445→        if (res.ok) {
   446→          const data = await res.json()

   451→              SOLUSDT: data.prices.SOLUSDT?.price || 0,

   454→            if (!wsConnectedRef.current) {

   462→    }

   468→        fetchPrices()
   469→      }
   470→    }, 5000)
   471→
   472→    return () => clearInterval(interval)
   473→  }, [mounted, updatePriceHistory])
   474→
   475→  // --- REST API round info polling (fallback when WS not connected, and no active bet) ---
   476→  useEffect(() => {
   477→    if (!mounted || wsConnected || activeBet) return
   478→
   479→    const pollRoundInfo = async () => {
   480→      const key = roundSubRef.current
   481→      if (!key) return
   482→      const [coin, durStr] = key.split("-")
   483→      const duration = parseInt(durStr, 10) as Duration
   484→      if (coin && duration) {
   485→        await fetchRoundInfo(coin as Coin, duration)
   486→      }
   487→    }
   488→
   489→    // Poll every 3 seconds for round info (when no active bet, for the predict screen timer)
   490→    const interval = setInterval(pollRoundInfo, 3000)
   491→    return () => clearInterval(interval)
   492→  }, [mounted, wsConnected, activeBet, fetchRoundInfo])
   493→
   494→  // --- Initialize WebSocket ---
   495→  useEffect(() => {

   497→
   498→    const socket = socketIO("/?XTransformPort=3003", {
   499→      transports: ["websocket"],
   500→      reconnection: true,
   501→      reconnectionAttempts: Infinity,
   502→      reconnectionDelay: 1000,
   503→    })
   504→
   505→    socketRef.current = socket
   506→
   507→    socket.on("connect", () => {

   509→      setWsConnected(true)
   510→    })
   511→
   512→    socket.on("disconnect", () => {

   514→      setWsConnected(false)
   515→    })
   516→

   518→      const newPrices: Record<Coin, number> = {
   519→        BTCUSDT: data.BTCUSDT,
   520→        ETHUSDT: data.ETHUSDT,
   521→        SOLUSDT: data.SOLUSDT,
   522→      }
   523→      setPrices(newPrices)
   524→      updatePriceHistory(newPrices)
   525→    })
   526→
   527→    socket.on("history:update", (data) => {
   528→      setPriceHistory({
   529→        BTCUSDT: data.BTCUSDT || [],
   530→        ETHUSDT: data.ETHUSDT || [],
   531→        SOLUSDT: data.SOLUSDT || [],
   532→      })
   533→    })
   534→
   535→    socket.on("round:tick", (data) => {
   536→      const key = `${data.coin}-${data.duration}`
   537→      if (key === roundSubRef.current) {
   538→        setCurrentRound({
   539→          roundId: data.roundId,
   540→          coin: data.coin as Coin,
   541→          duration: data.duration,
   542→          timeLeft: data.timeLeft,
   543→          entryPrice: data.entryPrice,
   544→          currentPrice: data.currentPrice || data.entryPrice,
   545→        })
   546→      }
   547→    })
   548→
   549→    socket.on("round:end", (data) => {
   550→      const key = `${data.coin}-${data.duration}`
   551→      if (key === roundSubRef.current) {
   552→        const currentActiveBet = activeBetRef.current
   553→
   554→        if (currentActiveBet && currentActiveBet.round.roundId === data.roundId) {
   555→          const wentUp = data.exitPrice > data.entryPrice
   556→          const won =
   557→            (currentActiveBet.direction === "UP" && wentUp) ||
   558→            (currentActiveBet.direction === "DOWN" && !wentUp)
   559→          const payout = won ? currentActiveBet.potentialPayout : 0
   560→
   561→          setLastResult({
   562→            roundId: data.roundId,
   563→            coin: data.coin,
   564→            direction: currentActiveBet.direction,
   565→            entryPrice: data.entryPrice,
   566→            exitPrice: data.exitPrice,
   567→            wentUp,
   568→            won,
   569→            amount: currentActiveBet.amount,
   570→            payout,
   571→          })
   572→
   573→          const c = countryRef.current
   574→          if (won) {
   575→            setWinEvent({
   576→              amount: currentActiveBet.potentialPayout,
   577→              direction: currentActiveBet.direction,
   578→              coin: data.coin,
   579→            })
   580→            pushToast(
   581→              `You WON ${c.symbol}${currentActiveBet.potentialPayout.toLocaleString(c.locale)}! 🎉`,
   582→              "win",
   583→            )
   584→          } else {
   585→            pushToast("Round lost — try again 💪", "lose")
   586→          }
   587→          setActiveBet(null)
   588→        }
   589→
   590→        // Refresh user data from server
   591→        refreshUser()
   592→        refreshBets()
   593→        refreshTransactions()
   594→      }
   595→    })
   596→
   597→    socket.on("round:new", (data) => {
   598→      const key = `${data.coin}-${data.duration}`
   599→      if (key === roundSubRef.current) {
   600→        setCurrentRound({
   601→          roundId: data.roundId,
   602→          coin: data.coin as Coin,
   603→          duration: data.duration,
   604→          timeLeft: data.timeLeft,
   605→          entryPrice: data.entryPrice,
   606→          currentPrice: data.entryPrice,
   607→        })
   608→      }
   609→    })
   610→
   611→    return () => {
   612→      socket.disconnect()
   613→    }
   614→  }, [mounted, updatePriceHistory, pushToast, refreshUser, refreshBets, refreshTransactions])
   615→
   616→  // --- Load initial data once userId is available ---
   617→  useEffect(() => {
   618→    if (!userId) return
   619→    const load = async () => {
   620→      await refreshActiveBet()
   621→      await refreshBets()
   622→      await refreshTransactions()
   623→    }
   624→    load()
   625→  }, [userId, refreshActiveBet, refreshBets, refreshTransactions])
   626→
   627→  // --- Check for round resolution when active bet exists but WS is not connected ---
   628→  useEffect(() => {

   648→          const r = data.result

   650→            roundId: r.roundId,
   651→            coin: r.coin,

   668→          } else {
   669→            pushToast("Round lost — try again 💪", "lose")
   670→          }
   671→          setActiveBet(null)
   672→          setCurrentRound(null)
   673→
   674→          // Refresh all data
   675→          await refreshUser()
   676→          await refreshBets()
   677→          await refreshTransactions()
   678→        } else if (data.activeBet && data.timeLeft !== undefined) {

   680→          setActiveBet(data.activeBet)
   681→          setCurrentRound({
   682→            roundId: data.activeBet.round.roundId,

   688→          })
   689→        }
   690→      } catch {
   691→        // Silent fail
   692→      }
   693→    }
   694→
   695→    // Check every 2 seconds when we have an active bet but no WS
   696→    const interval = setInterval(checkBetStatus, 2000)
   697→    return () => clearInterval(interval)
   698→  }, [mounted, wsConnected, activeBet, refreshUser, refreshBets, refreshTransactions, pushToast, country])

   700→  // --- Place Bet (API call - server resolves round) ---

   724→              entryPrice: roundInfo.entryPrice,
   725→            } : {}),
   726→          }),

   729→        const data = await res.json()
   730→
   731→        if (!res.ok) {
   732→          pushToast(data.error || "Failed to place bet", "lose")
   733→          return { success: false, error: data.error, code: data.code }
   734→        }
   735→

   737→        setBalance(data.balance)
   738→        if (data.dailyPredictions !== undefined) setDailyPredictions(data.dailyPredictions)
   739→        if (data.dailyLimit !== undefined) setDailyLimit(data.dailyLimit)
   740→        if (data.tier) setTier(data.tier)
   741→        setActiveBet({
   742→          id: data.id,
   743→          direction: data.direction,
   744→          amount: data.amount,
   745→          multiplier: data.multiplier,
   746→          potentialPayout: data.potentialPayout,
   747→          round: {
   748→            roundId: data.roundId,
   749→            coin: data.coin,
   750→            duration: data.duration,
   751→            entryPrice: data.entryPrice,
   752→          },
   753→          createdAt: data.createdAt,
   754→        })
   755→
   756→        pushToast(

   760→
   761→        // Refresh transactions
   762→        refreshTransactions()
   763→
   764→        // Subscribe to round updates
   765→        subscribeToRound(coin, duration)
   766→
   767→        return { success: true }
   768→      } catch {
   769→        pushToast("Network error — please try again", "lose")
   770→        return { success: false, error: "Network error" }
   771→      }
   772→    },
   773→    [userId, country, pushToast, refreshTransactions, subscribeToRound, fetchRoundInfo],
   774→  )

   949→export function timeLabels(count = 6) {
   950→  const now = new Date()
   951→  const labels: string[] = []
   952→  for (let i = count - 1; i >= 0; i--) {
   953→    const d = new Date(now.getTime() - i * 60_000)
   954→    labels.push(`${String(d.getHours()).padStart(2, "0")}:${String(d.getMinutes()).padStart(2, "0")}`)
   955→  }
   956→  return labels
   957→}

   959→export function priceLabels(history: number[], count = 7) {
   960→  if (history.length === 0) return Array.from({ length: count }, () => "--.--")
   961→  const min = Math.min(...history)
   962→  const max = Math.max(...history)
   963→  const pad = (max - min) * 0.1 || 100
   964→  const lo = min - pad
   965→  const hi = max + pad
   966→  const labels: string[] = []
   967→  for (let i = 0; i < count; i++) {
   968→    const v = hi - ((hi - lo) / (count - 1)) * i
   969→    labels.push(v.toLocaleString("en-US", { minimumFractionDigits: 2, maximumFractionDigits: 2 }))
   970→  }
   971→  return labels
   972→}
