   216→   216→export function CryptoProvider({ children }: { children: ReactNode }) {
   217→   217→  const mounted = useHydrated()
   218→   218→  const [country, setCountry] = useState<Country>(COUNTRIES[0])
   219→   219→
   220→   220→  // User
   221→   221→  const [userId, setUserId] = useState<string | null>(null)
   222→   222→  const [userName, setUserName] = useState("Trader")
   223→   223→  const [balance, setBalance] = useState(0)
   224→   224→  const [winnings, setWinnings] = useState(0)
   225→   225→  const [profit, setProfit] = useState(0)
   226→   226→  const [bonus, setBonus] = useState(0)
   227→   227→  const [wins, setWins] = useState(0)
   228→   228→  const [totalBets, setTotalBets] = useState(0)
   229→   229→  const [winRate, setWinRate] = useState(0)
   230→   230→
   231→   231→  // Premium tier
   232→   232→  const [tier, setTier] = useState<"FREE" | "PREMIUM">("FREE")
   233→   233→  const [dailyPredictions, setDailyPredictions] = useState(0)
   234→   234→  const [dailyLimit, setDailyLimit] = useState(3)
   235→   235→  const [premiumExpiresAt, setPremiumExpiresAt] = useState<string | null>(null)
   236→   236→
   237→   237→  // Prices from WebSocket or REST
   238→   238→  const [prices, setPrices] = useState<Record<Coin, number>>({
   239→   239→    BTCUSDT: 0,
   240→   240→    ETHUSDT: 0,
   241→   241→    SOLUSDT: 0,
   242→   242→  })
   243→   243→  const [priceHistory, setPriceHistory] = useState<Record<Coin, number[]>>({
   244→   244→    BTCUSDT: [],
   245→   245→    ETHUSDT: [],
   246→   246→    SOLUSDT: [],
   247→   247→  })
   248→   248→  const [currentRound, setCurrentRound] = useState<RoundInfo | null>(null)
   249→   249→  const [wsConnected, setWsConnected] = useState(false)
   250→   250→
   251→   251→  // Bet state
   252→   252→  const [activeBet, setActiveBet] = useState<ActiveBet>(null)
   253→   253→  const [recentBets, setRecentBets] = useState<BetHistory[]>([])
   254→   254→  const [transactions, setTransactions] = useState<Transaction[]>([])
   255→   255→
   256→   256→  // Round result (entry vs exit price comparison)
   257→   257→  const [lastResult, setLastResult] = useState<RoundResult>(null)
   258→   258→  const clearLastResult = useCallback(() => setLastResult(null), [])
   259→   259→
   260→   260→  // Win event
   261→   261→  const [winEvent, setWinEvent] = useState<WinEvent>(null)
   262→   262→  const clearWinEvent = useCallback(() => setWinEvent(null), [])
   263→   263→
   264→   264→  // Toasts
   265→   265→  const [toasts, setToasts] = useState<Toast[]>([])
   266→   266→  const pushToast = useCallback((msg: string, tone: Toast["tone"]) => {
   267→   267→    const id = Date.now() + Math.random()
   268→   268→    setToasts((t) => [...t, { id, msg, tone }])
   269→   269→    setTimeout(() => setToasts((t) => t.filter((x) => x.id !== id)), 3500)
   270→   270→  }, [])
   271→   271→
   272→   272→  // Refs to avoid stale closures
   273→   273→  const socketRef = useRef<Socket | null>(null)
   274→   274→  const roundSubRef = useRef<string>("")
   275→   275→  const activeBetRef = useRef<ActiveBet>(null)
   276→   276→  const userIdRef = useRef<string | null>(null)
   277→   277→  const countryRef = useRef<Country>(country)
   278→   278→  const wsConnectedRef = useRef(false)
   279→   279→  const currentRoundRef = useRef<RoundInfo | null>(null)
   280→   280→
   281→   281→  // Keep refs in sync
   282→   282→  useEffect(() => { activeBetRef.current = activeBet }, [activeBet])
   283→   283→  useEffect(() => { userIdRef.current = userId }, [userId])
   284→   284→  useEffect(() => { countryRef.current = country }, [country])
   285→   285→  useEffect(() => { wsConnectedRef.current = wsConnected }, [wsConnected])
   286→   286→  useEffect(() => { currentRoundRef.current = currentRound }, [currentRound])
   287→   287→
   288→   288→  // --- Helper: Update price history from any source ---
   289→   289→  const updatePriceHistory = useCallback((newPrices: Record<Coin, number>) => {
   290→   290→    setPriceHistory((prev) => {
   291→   291→      const updated = { ...prev }
   292→   292→      for (const coin of ["BTCUSDT", "ETHUSDT", "SOLUSDT"] as Coin[]) {
   293→   293→        if (newPrices[coin] > 0) {
   294→   294→          const arr = [...(prev[coin] || []), newPrices[coin]]
   295→   295→          if (arr.length > MAX_HISTORY) arr.shift()
   296→   296→          updated[coin] = arr
   297→   297→        }
   298→   298→      }
   299→   299→      return updated
   300→   300→    })
   301→   301→  }, [])
   302→   302→
   303→   303→  // --- Initialize User ---
   304→   304→  useEffect(() => {
   305→   305→    if (!mounted) return
   306→   306→    fetch("/api/user")
   307→   307→      .then((r) => r.json())
   308→   308→      .then((data) => {
   309→   309→        setUserId(data.id)
   310→   310→        setUserName(data.name)
   311→   311→        setBalance(data.balance)
   312→   312→        setWinnings(data.winnings)
   313→   313→        setProfit(data.profit)
   314→   314→        setBonus(data.bonus)
   315→   315→        setWins(data.wins)
   316→   316→        setTotalBets(data.totalBets)
   317→   317→        setWinRate(data.winRate)
   318→   318→        setTier(data.tier || "FREE")
   319→   319→        setDailyPredictions(data.dailyPredictions || 0)
   320→   320→        setDailyLimit(data.dailyLimit ?? 3)
   321→   321→        setPremiumExpiresAt(data.premiumExpiresAt || null)
   322→   322→      })
   323→   323→      .catch(console.error)
   324→   324→  }, [mounted])
   325→   325→
   326→   326→  // --- Subscribe to a specific coin+duration round ---
   327→   327→  const subscribeToRound = useCallback((coin: Coin, duration: Duration) => {
   328→   328→    const key = `${coin}-${duration}`
   329→   329→    roundSubRef.current = key
   330→   330→    socketRef.current?.emit("round:subscribe", { coin, duration })
   331→   331→  }, [])
   332→   332→
   333→   333→  // --- Fetch round info via REST API (works without WebSocket) ---

   335→   335→    try {
   336→   336→      const res = await fetch(`/api/round/current?coin=${coin}&duration=${duration}`)
   337→   337→      if (res.ok) {
   338→   338→        const data = await res.json()
   339→   339→        const info: RoundInfo = {
   340→   340→          roundId: data.roundId,
   341→   341→          coin: data.coin as Coin,
   342→   342→          duration: data.duration,
   343→   343→          timeLeft: data.timeLeft,
   344→   344→          entryPrice: data.entryPrice,
   345→   345→          currentPrice: data.currentPrice || data.entryPrice,
   346→   346→        }
   347→   347→        // Update current round state
   348→   348→        setCurrentRound(info)
   349→   349→        return info
   350→   350→      }
   351→   351→    } catch {
   352→   352→      // Silently fail
   353→   353→    }
   354→   354→    return null
   355→   355→  }, [])
   356→   356→
   357→   357→  // --- Fetch active bet ---

   359→   359→    const uid = userIdRef.current
   360→   360→    if (!uid) return

   362→   362→      const res = await fetch(`/api/bet/active?userId=${uid}`)

   370→   370→          )

   372→   372→          fetchRoundInfo(
   373→   373→            data.activeBet.round.coin as Coin,
   374→   374→            data.activeBet.round.duration as Duration,
   375→   375→          )
   376→   376→        }
   377→   377→      }
   378→   378→    } catch (err) {
   379→   379→      console.error("Failed to refresh active bet:", err)
   380→   380→    }
   381→   381→  }, [subscribeToRound, fetchRoundInfo])
   382→   382→
   383→   383→  // --- Refresh user data ---
   384→   384→  const refreshUser = useCallback(async () => {
   385→   385→    const uid = userIdRef.current
   386→   386→    if (!uid) return
   387→   387→    try {
   388→   388→      const res = await fetch("/api/user")
   389→   389→      if (res.ok) {
   390→   390→        const data = await res.json()
   391→   391→        setBalance(data.balance)
   392→   392→        setWinnings(data.winnings)
   393→   393→        setProfit(data.profit)
   394→   394→        setBonus(data.bonus)
   395→   395→        setWins(data.wins)
   396→   396→        setTotalBets(data.totalBets)
   397→   397→        setWinRate(data.winRate)
   398→   398→        setTier(data.tier || "FREE")
   399→   399→        setDailyPredictions(data.dailyPredictions || 0)
   400→   400→        setDailyLimit(data.dailyLimit ?? 3)
   401→   401→        setPremiumExpiresAt(data.premiumExpiresAt || null)
   402→   402→      }
   403→   403→    } catch (err) {
   404→   404→      console.error("Failed to refresh user:", err)
   405→   405→    }
   406→   406→  }, [])
   407→   407→
   408→   408→  // --- Refresh bet history ---
   409→   409→  const refreshBets = useCallback(async () => {
   410→   410→    const uid = userIdRef.current
   411→   411→    if (!uid) return

   413→   413→      const res = await fetch(`/api/bets?userId=${uid}`)
   414→   414→      if (res.ok) {
   415→   415→        const data = await res.json()
   416→   416→        setRecentBets(data.bets || [])
   417→   417→      }
   418→   418→    } catch (err) {
   419→   419→      console.error("Failed to refresh bets:", err)
   420→   420→    }
   421→   421→  }, [])
   422→   422→
   423→   423→  // --- Refresh transactions ---
   424→   424→  const refreshTransactions = useCallback(async () => {
   425→   425→    const uid = userIdRef.current
   426→   426→    if (!uid) return

   428→   428→      const res = await fetch(`/api/transactions?userId=${uid}`)

   430→   430→        const data = await res.json()
   431→   431→        setTransactions(data.transactions || [])

   433→   433→    } catch (err) {
   434→   434→      console.error("Failed to refresh transactions:", err)
   435→   435→    }
   436→   436→  }, [])
   437→   437→

   439→   439→  useEffect(() => {
   440→   440→    if (!mounted) return
   441→   441→
   442→   442→    const fetchPrices = async () => {
   443→   443→      try {
   444→   444→        const res = await fetch("/api/prices")
   445→   445→        if (res.ok) {
   446→   446→          const data = await res.json()
   447→   447→          if (data.prices) {
   448→   448→            const newPrices: Record<Coin, number> = {
   449→   449→              BTCUSDT: data.prices.BTCUSDT?.price || 0,
   450→   450→              ETHUSDT: data.prices.ETHUSDT?.price || 0,
   451→   451→              SOLUSDT: data.prices.SOLUSDT?.price || 0,
   452→   452→            }
   453→   453→            setPrices(newPrices)
   454→   454→            if (!wsConnectedRef.current) {
   455→   455→              updatePriceHistory(newPrices)
   456→   456→            }
   457→   457→          }
   458→   458→        }
   459→   459→      } catch {
   460→   460→        // Silent fail
   461→   461→      }
   462→   462→    }
   463→   463→
   464→   464→    fetchPrices()
   465→   465→
   466→   466→    const interval = setInterval(() => {
   467→   467→      if (!wsConnectedRef.current) {
   468→   468→        fetchPrices()
   469→   469→      }
   470→   470→    }, 5000)
   471→   471→
   472→   472→    return () => clearInterval(interval)
   473→   473→  }, [mounted, updatePriceHistory])
   474→   474→
   475→   475→  // --- REST API round info polling (fallback when WS not connected, and no active bet) ---
   476→   476→  useEffect(() => {
   477→   477→    if (!mounted || wsConnected || activeBet) return
   478→   478→
   479→   479→    const pollRoundInfo = async () => {
   480→   480→      const key = roundSubRef.current
   481→   481→      if (!key) return
   482→   482→      const [coin, durStr] = key.split("-")
   483→   483→      const duration = parseInt(durStr, 10) as Duration
   484→   484→      if (coin && duration) {
   485→   485→        await fetchRoundInfo(coin as Coin, duration)
   486→   486→      }
   487→   487→    }
   488→   488→
   489→   489→    // Poll every 3 seconds for round info (when no active bet, for the predict screen timer)
   490→   490→    const interval = setInterval(pollRoundInfo, 3000)
   491→   491→    return () => clearInterval(interval)
   492→   492→  }, [mounted, wsConnected, activeBet, fetchRoundInfo])
   493→   493→
   494→   494→  // --- Initialize WebSocket ---
   495→   495→  useEffect(() => {
   496→   496→    if (!mounted) return
   497→   497→
   498→   498→    const socket = socketIO("/?XTransformPort=3003", {
   499→   499→      transports: ["websocket"],
   500→   500→      reconnection: true,

   949→   949→export function timeLabels(count = 6) {
   950→   950→  const now = new Date()
   951→   951→  const labels: string[] = []
   952→   952→  for (let i = count - 1; i >= 0; i--) {
   953→   953→    const d = new Date(now.getTime() - i * 60_000)
   954→   954→    labels.push(`${String(d.getHours()).padStart(2, "0")}:${String(d.getMinutes()).padStart(2, "0")}`)
   955→   955→  }
   956→   956→  return labels

   959→   959→export function priceLabels(history: number[], count = 7) {
   960→   960→  if (history.length === 0) return Array.from({ length: count }, () => "--.--")
   961→   961→  const min = Math.min(...history)
   962→   962→  const max = Math.max(...history)
   963→   963→  const pad = (max - min) * 0.1 || 100
   964→   964→  const lo = min - pad
   965→   965→  const hi = max + pad
   966→   966→  const labels: string[] = []
   967→   967→  for (let i = 0; i < count; i++) {
   968→   968→    const v = hi - ((hi - lo) / (count - 1)) * i
   969→   969→    labels.push(v.toLocaleString("en-US", { minimumFractionDigits: 2, maximumFractionDigits: 2 }))
   970→   970→  }
   971→   971→  return labels
   972→   972→}
