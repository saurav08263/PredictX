    34→    34→ *
    35→    35→ * ONE screen shows everything:
    36→    36→ *   - Coin tabs (BTC/ETH/SOL) — compact horizontal pills

    55→    55→

   143→   143→      setBetPlacing(false)
   144→   144→    }

   174→   174→    }
   175→   175→  }

   194→   194→            UNLIMITED →

   199→   199→      {/* Premium badge for premium users */}
   200→   200→      {tier === "PREMIUM" && effectiveStep === "trade" && (
   201→   201→        <div className="flex items-center justify-center gap-1.5 rounded-lg border border-gold/40 bg-gold/10 px-3 py-1.5">
   202→   202→          <Zap className="size-3 text-gold" />
   203→   203→          <span className="text-[10px] font-bold text-gold">PREMIUM · UNLIMITED PREDICTIONS</span>

   205→   205→      )}

   210→   210→          <motion.div
   211→   211→            key="trade"
   212→   212→            initial={{ opacity: 0 }}
   213→   213→            animate={{ opacity: 1 }}
   214→   214→            exit={{ opacity: 0 }}
   215→   215→            className="flex flex-col gap-2"
   216→   216→          >

   277→   277→                      <span className="size-1.5 rounded-full bg-bull animate-pulse" />LIVE

   280→   280→                  {currentRound && (
   281→   281→                    <span className="flex items-center gap-1 rounded-md border border-border bg-muted px-1.5 py-0.5 text-[9px] font-bold tabular-nums">

   289→   289→              {/* Chart */}
